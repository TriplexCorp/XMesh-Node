func TestCanonicalUserDenyRecoveredAcrossSignedInboundAliases(t *testing.T) {
    main := "xmesh-u101--vless-share"
    cdn := "xmesh-u101--vless-cdn"
    if durableFencePath("/tmp",main) != durableFencePath("/tmp",cdn) ||
        checkpointPath("/tmp",main) != checkpointPath("/tmp",cdn) {
        t.Fatal("one signed user can create separate main and CDN revocation slots")
    }
    dir,_ := setupCheckpoint(t,main,0,0)
    // setupCheckpoint persists the primary email's signed-scope fence.
    before,err := readDurableFence(durableFencePath(dir,main))
    if err!=nil || before.Email!=main {t.Fatalf("untrusted durable fence: %v",err)}
    resetGate()
    if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
    if err:=Register(ctx(901),cdn,func(){});!errors.Is(err,errFenced){
        t.Fatalf("restarted Xray admitted same user via alternate CDN email: %v",err)
    }
    if err:=Register(ctx(902),"xmesh-u102--vless-cdn",func(){});err!=nil{
        t.Fatalf("unrelated billing user was blocked: %v",err)
    }
    altered:=testScope(cdn,testRevocation)
    if err:=commitDurableFence(altered);err==nil{
        t.Fatal("different CDN email overwrote a stable user fence")
    }
}

func TestCrashBeforeFenceHardlinkRefusesRestart(t *testing.T) {
    dir,_:=setupCheckpoint(t,"power-loss-before-link",0,0)
    marker:=durableFencePath(dir,"power-loss-before-link")
    data,err:=os.ReadFile(marker)
    if err!=nil{t.Fatal(err)}
    // Simulate fsynced .fence-pending-* but NO published fence-*.json.
    pending:=filepath.Join(dir,".fence-pending-power-loss")
    if err:=os.WriteFile(pending,data,0600);err!=nil{t.Fatal(err)}
    if err:=os.Remove(marker);err!=nil{t.Fatal(err)}
    resetGate()
    if err:=configureObservedCheckpoint(dir);err==nil {
        t.Fatal("unpublished signed revocation intent was silently forgotten at restart")
    }
    if _,err:=os.Lstat(pending);err!=nil{
        t.Fatalf("orphan pending fence was silently discarded: %v",err)
    }
}

func TestCrashAfterFenceHardlinkRepairsAndRestoresDeny(t *testing.T) {
    dir,_:=setupCheckpoint(t,"power-loss-after-link",0,0)
    marker:=durableFencePath(dir,"power-loss-after-link")
    pending:=filepath.Join(dir,".fence-pending-postlink")
    if err:=os.Link(marker,pending);err!=nil{t.Fatal(err)}
    resetGate()
    if err:=configureObservedCheckpoint(dir);err!=nil{
        t.Fatalf("replay of already published hardlink must be recoverable: %v",err)
    }
    if _,err:=os.Lstat(pending);!errors.Is(err,os.ErrNotExist){
        t.Fatalf("reconciled pending hardlink remains after fsync: %v",err)
    }
    if _,err:=os.Lstat(marker);err!=nil{t.Fatal(err)}
    if err:=Register(ctx(202), "power-loss-after-link",func(){});!errors.Is(err,errFenced){
        t.Fatalf("reconciled tombstone failed to block old user: %v",err)
    }
    if err:=Register(ctx(203), "healthy-other-user",func(){});err!=nil {
        t.Fatalf("reconciliation denied an unrelated user: %v",err)
    }
}

func TestUnpublishedOrForgedPendingObservedJournalRefusesRestart(t *testing.T) {
    for _,scenario:=range []string{"orphaned-observation","different-inode","symlink-pending"} {
        t.Run(scenario,func(t *testing.T) {
            dir,_:=setupCheckpoint(t,"meter-pending-"+scenario,4,5)
            scope:=testScope("meter-pending-"+scenario,testRevocation)
            if _,err:=persistObservedCheckpoint(scope);err!=nil{t.Fatal(err)}
            original:=checkpointPath(dir,scope.Email)
            pending:=filepath.Join(dir,".observed-pending-suspicious")
            switch scenario {
            case "orphaned-observation":
                if err:=os.Link(original,pending);err!=nil{t.Fatal(err)}
                if err:=os.Remove(original);err!=nil{t.Fatal(err)}
            case "different-inode":
                b,err:=os.ReadFile(original)
                if err!=nil{t.Fatal(err)}
                if err:=os.WriteFile(pending,b,0600);err!=nil{t.Fatal(err)}
            case "symlink-pending":
                if err:=os.Symlink(original,pending);err!=nil{t.Fatal(err)}
            }
            resetGate()
            if err:=configureObservedCheckpoint(dir);err==nil {
                t.Fatalf("restarted after unsafe pending journal: %s",scenario)
            }
            if _,err:=os.Lstat(pending);err!=nil{
                t.Fatalf("unsafe pending evidence was silently discarded: %v",err)
            }
        })
    }
}

func TestDurableFenceRecoveredBeforeAdmission(t *testing.T) {
    dir,_:=setupCheckpoint(t,"persistent",0,0)
    scope:=testScope("persistent",testRevocation)
    path:=durableFencePath(dir,scope.Email)
    disk,err:=readDurableFence(path)
    if err!=nil||disk.checkpointScope!=scope||disk.SchemaVersion!=1 {
        t.Fatalf("bad durable scope: %+v %v",disk,err)
    }
    if stat,err:=os.Lstat(path);err!=nil||stat.Mode().Perm()&0077!=0 {
        t.Fatalf("unsafe durable marker perms: %v %v",stat,err)
    }
    resetGate()
    // Simulate the Xray process boundary. Any old in-memory gate is gone.
    if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
    if err:=Register(ctx(172),"persistent",func(){});!errors.Is(err,errFenced) {
        t.Fatalf("recovered revoked user admitted: %v",err)
    }
    if err:=Register(ctx(173),"unrelated",func(){});err!=nil {
        t.Fatalf("unrelated user incorrectly denied: %v",err)
    }
    if err:=commitDurableFence(scope);err!=nil{
        t.Fatalf("identical replay incorrectly rejected: %v",err)
    }
    altered:=scope
    altered.Revision++
    if err:=commitDurableFence(altered);err==nil{
        t.Fatal("cross-revision replay overwrote an immutable fence")
    }
    got,err:=readDurableFence(path)
    if err!=nil||got!=disk{t.Fatalf("durable scope mutated: %+v %v",got,err)}
}

func TestDurableFenceRejectsInvalidSymlinkAndTamperedScopes(t *testing.T) {
    for _,kind:=range []string{"symlink","world-readable","wrong-name","wrong-schema"} {
        t.Run(kind,func(t *testing.T){
            resetGate()
            dir:=t.TempDir()
            if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
            record:=durableFenceRecord{
                checkpointScope:testScope("x",testRevocation),
                SchemaVersion:1,
            }
            data,err:=json.Marshal(record)
            if err!=nil{t.Fatal(err)}
            path:=durableFencePath(dir,"x")
            switch kind {
            case "symlink":
                victim:=filepath.Join(dir,"victim")
                if err:=os.WriteFile(victim,data,0600);err!=nil{t.Fatal(err)}
                if err:=os.Symlink(victim,path);err!=nil{t.Fatal(err)}
            case "world-readable":
                if err:=os.WriteFile(path,data,0644);err!=nil{t.Fatal(err)}
            case "wrong-name":
                path=filepath.Join(dir,"fence-"+strings.Repeat("0",64)+".json")
                if err:=os.WriteFile(path,data,0600);err!=nil{t.Fatal(err)}
            case "wrong-schema":
                record.SchemaVersion=3
                data,err=json.Marshal(record)
                if err!=nil{t.Fatal(err)}
                if err:=os.WriteFile(path,data,0600);err!=nil{t.Fatal(err)}
            }
            if err:=configureObservedCheckpoint(dir);err==nil {
                t.Fatalf("recovered invalid %s tombstone",kind)
            }
        })
    }
}

// A syntactically valid JSON object is not a safe crash-recovery fence.
// Duplicate keys, unknown additions and alternate encodings might otherwise
// hide a different lease or bypass downstream byte-identical comparisons.
func TestNoncanonicalDurableFenceRefusesXrayRestart(t *testing.T) {
    email:="xmesh-u101--vless-main"
    for _,kind:=range []string{
        "duplicate-scope","duplicate-schema","unknown-field",
        "trailing-blank","read-only-mode",
    } {
        t.Run(kind,func(t *testing.T) {
            dir,_:=setupCheckpoint(t,email,3,4)
            path:=durableFencePath(dir,email)
            b,err:=os.ReadFile(path)
            if err!=nil{t.Fatal(err)}
            data:=string(b)
            switch kind {
            case "duplicate-scope":
                data=strings.Replace(data,`"nodeId":201`,
                    `"nodeId":201,"nodeId":201`,1)
            case "duplicate-schema":
                data=strings.Replace(data,`"schemaVersion":1`,
                    `"schemaVersion":1,"schemaVersion":1`,1)
            case "unknown-field":
                data=strings.Replace(data,`"schemaVersion":1`,
                    `"schemaVersion":1,"finalMeterVerified":true`,1)
            case "trailing-blank":
                data+=" "
            case "read-only-mode":
                if err:=os.Chmod(path,0400);err!=nil{t.Fatal(err)}
            }
            if kind!="read-only-mode" {
                if data==string(b){t.Fatal("tamper case failed to alter journal")}
                if err:=os.WriteFile(path,[]byte(data),0600);err!=nil{t.Fatal(err)}
            }
            if _,err:=readDurableFence(path);err==nil {
                t.Fatal("noncanonical admission fence was accepted as durable")
            }
            resetGate()
            if err:=configureObservedCheckpoint(dir);err==nil {
                t.Fatal("corrupt durable denial silently allowed Xray restart")
            }
        })
    }
}

func TestDurableFencePersistenceFailureNeverStopsAsAcknowledged(t *testing.T) {
    resetGate()
    dir:=t.TempDir()
    if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
    if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
    scope:=testScope("denied-on-error",testRevocation)
    id:=ctx(742)
    if err:=Register(id,scope.Email,func(){});err!=nil{t.Fatal(err)}
    if err:=os.Chmod(dir,0755);err!=nil{t.Fatal(err)}
    _,err:=fenceScoped(context.Background(),scope)
    if err==nil {t.Fatal("unsafe dir returned an acknowledged fence")}
    gate.Lock()
    u:=gate.users[scope.Email]
    fenced:=u.fenced
    gate.Unlock()
    if fenced {t.Fatal("fence was claimed after durability precondition failed")}
    if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
    Finish(id)
    RouteFinish(id)
}

func TestRuntimeLockAndStaleSocketSafety(t *testing.T) {
    dir:=t.TempDir()
    if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
    if err:=lockExperimentalInstance(dir);err!=nil{t.Fatal(err)}
    defer func(){
        if instanceLock!=nil{
            _=instanceLock.Close()
            instanceLock=nil
        }
    }()
    if err:=lockExperimentalInstance(dir);err==nil {
        t.Fatal("multiple live Xray processes can acquire the same private journal")
    }
    endpoint:=filepath.Join(dir,"control.sock")
    listener,err:=net.Listen("unix",endpoint)
    if err!=nil{t.Fatal(err)}
    if err:=recoverStaleUnixSocket(endpoint);err==nil {
        _=listener.Close()
        t.Fatal("would unlink a live privileged control socket")
    }
    if unix,ok:=listener.(*net.UnixListener);ok {
        unix.SetUnlinkOnClose(false)
    }
    if err:=listener.Close();err!=nil{t.Fatal(err)}
    if err:=recoverStaleUnixSocket(endpoint);err!=nil{
        t.Fatalf("SIGKILL-style stale socket was not recoverable: %v",err)
    }
    if _,err:=os.Lstat(endpoint);!errors.Is(err,os.ErrNotExist){
        t.Fatalf("stale control socket not unlinked: %v",err)
    }
    if err:=os.Symlink("victim",endpoint);err!=nil{t.Fatal(err)}
    if err:=recoverStaleUnixSocket(endpoint);err==nil {
        t.Fatal("unsafe socket symlink was removed")
    }
}
