// CI-ONLY Xray observed-meter checkpoint. Not a terminal byte proof.
// This file is injected exclusively into a disposable pinned Xray checkout.
// A process can finish a routed task while outbound goroutines still write:
// the observed counters must NEVER authorize quota refunds.
package xmeshdrain

import (
    "crypto/rand"
    "crypto/sha256"
    "encoding/hex"
    "encoding/json"
    "errors"
    "fmt"
    "os"
    "path/filepath"
    "reflect"
    "sort"
    "sync"

    "github.com/xtls/xray-core/features/stats"
)

// An observational receipt is scoped to a lease, not just an Xray email.
// Caller-supplied scope is NOT a signed Node policy and never authorizes refunds.
type checkpointScope struct {
    Email string `json:"email"`
    NodeID int64 `json:"nodeId"`
    UserID int64 `json:"userId"`
    Epoch string `json:"epoch"`
    MeterEpoch string `json:"meterEpoch"`
    ReservationID string `json:"reservationId"`
    Revision int64 `json:"revision"`
    RevocationID string `json:"revocationId"`
}

type observedInboundMeter struct {
    Email string `json:"email"`
    UplinkBytes uint64 `json:"uplinkBytes"`
    DownlinkBytes uint64 `json:"downlinkBytes"`
}

type observedCheckpoint struct {
    checkpointScope
    SchemaVersion int `json:"schemaVersion"`
    RuntimeID string `json:"runtimeId"`
    UplinkBytes uint64 `json:"uplinkBytes"`
    DownlinkBytes uint64 `json:"downlinkBytes"`
    // Exact sorted raw per-inbound identities, NOT merely one combined
    // counter. Node can independently compare this to signed policy scope.
    InboundMeters []observedInboundMeter `json:"inboundMeters"`
    Drained bool `json:"drained"`
    // Deliberately false; callers MUST NOT treat the scope as signed proof.
    FinalMeterVerified bool `json:"finalMeterVerified"`
    RefundableBytes uint64 `json:"refundableBytes"`
}

var observed = struct {
    sync.Mutex
    manager stats.Manager
    directory string
    runtimeID string
}{}

// configureObservedCheckpoint generates a new runtime identity and requires
// a private, pre-existing directory. The real CI process calls this at start.
// Never infer a trusted runtime from a previous disk file.
func configureObservedCheckpoint(directory string) error {
    if directory == "" || !filepath.IsAbs(directory) {
        return errors.New("observed checkpoint directory must be absolute")
    }
    info,err := os.Lstat(directory)
    if err != nil || !info.IsDir() || info.Mode().Perm()&0077 != 0 {
        return errors.New("observed checkpoint directory must exist and be private")
    }
    var bytes [32]byte
    if _,err := rand.Read(bytes[:]); err != nil { return err }
    // A recovered durable fence must be installed before Xray admits users.
    if err:=restoreDurableFences(directory);err!=nil{return err}
    observed.Lock()
    defer observed.Unlock()
    observed.directory = directory
    observed.manager = nil // An old in-memory Xray process meter is never recoverable.
    observed.runtimeID = hex.EncodeToString(bytes[:])
    return nil
}

func BindStatsManager(manager stats.Manager) {
    if !enabled { return }
    observed.Lock()
    defer observed.Unlock()
    observed.manager = manager
}

func validateRevocationID(id string) bool {
    if len(id)!=36 { return false }
    for i,c := range id {
        if i==8 || i==13 || i==18 || i==23 {
            if c!='-' { return false }
        } else if !((c>='0'&&c<='9')||(c>='a'&&c<='f')) {
            return false
        }
    }
    return true
}

func validScope(scope checkpointScope) bool {
    if scope.Email=="" || len(scope.Email)>128 || scope.NodeID<=0 ||
        scope.UserID<=0 || scope.Revision<=0 { return false }
    ids:=[]string{scope.Epoch,scope.MeterEpoch,scope.ReservationID,scope.RevocationID}
    for _,id:=range ids {
        if !validateRevocationID(id) { return false }
        // A nil or all-zero UUID cannot identify a real lease or revocation.
        if id=="00000000-0000-0000-0000-000000000000" { return false }
    }
    return true
}

func checkpointPath(directory,email string) string {
    digest := sha256.Sum256([]byte(canonicalMeterUser(email)))
    return filepath.Join(directory,"observed-"+hex.EncodeToString(digest[:])+".json")
}

func loadCheckpoint(path string) (observedCheckpoint,error) {
    var existing observedCheckpoint
    info,err := os.Lstat(path)
    if err != nil { return existing,err }
    if !info.Mode().IsRegular() || info.Mode().Perm()&0077 != 0 ||
        info.Size() <= 0 || info.Size() > 4096 {
        return existing,errors.New("unsafe existing checkpoint file")
    }
    b,err := os.ReadFile(path)
    if err != nil { return existing,err }
    if err = json.Unmarshal(b,&existing); err != nil { return existing,err }
    if existing.SchemaVersion != 3 || !validScope(existing.checkpointScope) ||
        !existing.Drained ||
        existing.FinalMeterVerified || existing.RefundableBytes != 0 ||
        !validObservedInboundMeters(existing) {
        return existing,errors.New("unsupported, incomplete or refundable existing checkpoint")
    }
    return existing,nil
}

// Meter rows must be sorted, unique and exactly sum to the checkpoint
// totals. These are still just observational bytes, not terminal settlement.
func validObservedInboundMeters(record observedCheckpoint) bool {
    rows:=record.InboundMeters
    if len(rows)==0 || len(rows)>16{return false}
    const limit=uint64(1<<63-1)
    var uplink,downlink uint64
    previous:=""
    sawRequest:=false
    for _,row:=range rows {
        if row.Email=="" || len(row.Email)>128 ||
           canonicalMeterUser(row.Email)!=canonicalMeterUser(record.Email) ||
           (previous!="" && row.Email<=previous) ||
           row.UplinkBytes>limit-uplink || row.DownlinkBytes>limit-downlink {
            return false
        }
        if row.Email==record.Email{sawRequest=true}
        uplink+=row.UplinkBytes
        downlink+=row.DownlinkBytes
        previous=row.Email
    }
    return sawRequest && downlink<=limit-uplink &&
        uplink==record.UplinkBytes && downlink==record.DownlinkBytes
}

// persistObservedCheckpoint is called ONLY after fence() has stopped new
// admission and seen both worker and routedDispatch return. It samples the
// original Xray StatsService counters while Xray is still running. Such a
// sample remains nonterminal because not all writers have proven completion.
//
// A repeated request must have the identical (full lease scope, runtime,
// counters) tuple. A changed meter or process incarnation invalidates the
// old snapshot rather than silently overwriting/refunding it.
func persistObservedCheckpoint(scope checkpointScope) (observedCheckpoint,error) {
    var empty observedCheckpoint
    if !validScope(scope) {
        return empty,errors.New("invalid observed checkpoint lease scope")
    }
    email:=scope.Email
    observed.Lock()
    defer observed.Unlock()
    if observed.directory=="" || observed.runtimeID=="" || observed.manager==nil {
        return empty,errors.New("observed checkpoint not configured")
    }
    intent,err:=readDurableFence(durableFencePath(observed.directory,email))
    if err!=nil || intent.checkpointScope!=scope {
        return empty,errors.New("no matching durable pre-stop denial for observation")
    }
    // The fencing gate is independent of this manager lock. Recheck the
    // post-drain state, but never interpret it as final byte accounting.
    gate.Lock()
    u:=gate.users[canonicalMeterUser(email)]
    drained:=u!=nil && u.fenced && u.closed && len(u.active)==0 && u.writes==0 && u.reads==0
    var seenEmails []string
    if u!=nil {
        for alias:=range u.emails { seenEmails=append(seenEmails,alias) }
    }
    gate.Unlock()
    if !drained { return empty,errors.New("cannot sample undrained user") }
    if len(seenEmails)==0 || len(seenEmails)>16 {
        return empty,errors.New("missing or unbounded observed inbound identities")
    }
    sort.Strings(seenEmails)
    // Every observed account-bearing inbound must contribute to the sample.
    // A single request-email read would undercount actual main+CDN traffic.
    // Still NONFINAL: observed sessions are NOT an authenticated enumeration
    // of every provisioned signed inbound, and unrelated background writers
    // have not been proven quiescent.
    foundRequestEmail:=false
    var uplink,downlink uint64
    meters:=make([]observedInboundMeter,0,len(seenEmails))
    const maxSignedMeter=uint64(1<<63-1)
    read := func(alias,direction string)(uint64,error) {
        counter:=observed.manager.GetCounter("user>>>"+alias+">>>traffic>>>"+direction)
        if counter==nil { return 0,fmt.Errorf("missing %s counter for %s",direction,alias) }
        n:=counter.Value()
        if n<0 { return 0,fmt.Errorf("negative %s counter for %s",direction,alias) }
        return uint64(n),nil
    }
    for _,alias:=range seenEmails {
        if canonicalMeterUser(alias)!=canonicalMeterUser(email) {
            return empty,errors.New("observed alias switched billed user")
        }
        if alias==email { foundRequestEmail=true }
        up,err:=read(alias,"uplink")
        if err!=nil{return empty,err}
        down,err:=read(alias,"downlink")
        if err!=nil{return empty,err}
        if up>maxSignedMeter-uplink || down>maxSignedMeter-downlink {
            return empty,errors.New("observed inbound byte totals overflow")
        }
        uplink+=up
        downlink+=down
        meters=append(meters,observedInboundMeter{
            Email:alias,UplinkBytes:up,DownlinkBytes:down,
        })
    }
    if !foundRequestEmail || downlink>maxSignedMeter-uplink {
        return empty,errors.New("missing request email or observed total overflow")
    }
    record:=observedCheckpoint{
        checkpointScope:scope,
        SchemaVersion:3,
        RuntimeID:observed.runtimeID,UplinkBytes:uplink,
        DownlinkBytes:downlink,InboundMeters:meters,Drained:true,
        FinalMeterVerified:false,RefundableBytes:0,
    }
    if !validObservedInboundMeters(record) {
        return empty,errors.New("invalid observed inbound meter scope")
    }
    path:=checkpointPath(observed.directory,email)
    if old,err:=loadCheckpoint(path);err==nil{
        if !reflect.DeepEqual(old,record) {
            return empty,errors.New("existing checkpoint mismatch or late counter update")
        }
        return old,nil
    }else if !os.IsNotExist(err){
        return empty,fmt.Errorf("unsafe checkpoint: %w",err)
    }

    data,err:=json.Marshal(record)
    if err!=nil{return empty,err}
    temporary,err:=os.CreateTemp(observed.directory,".observed-pending-")
    if err!=nil{return empty,err}
    tempName:=temporary.Name()
    defer os.Remove(tempName)
    if _,err=temporary.Write(append(data,byte(10)));err!=nil{
        temporary.Close()
        return empty,err
    }
    if err=temporary.Sync();err!=nil{
        temporary.Close()
        return empty,err
    }
    if err=temporary.Close();err!=nil{return empty,err}
    // Link makes creation exclusive: never replace an earlier runtime's
    // observation or follow an attacker-supplied destination symlink.
    if err=os.Link(tempName,path);err!=nil{
        return empty,fmt.Errorf("checkpoint already present or unsafe: %w",err)
    }
    directory,err:=os.Open(observed.directory)
    if err!=nil{return empty,err}
    syncErr:=directory.Sync()
    closeErr:=directory.Close()
    if syncErr!=nil{return empty,syncErr}
    if closeErr!=nil{return empty,closeErr}
    return record,nil
}
