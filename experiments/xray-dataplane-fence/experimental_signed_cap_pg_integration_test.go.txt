// Cross-repository disposable integration. Real PostgreSQL quota allocator,
// real Ed25519 signature and real Xray StatsService/WAL in a live Go process.
// The standalone Python CI signer SIMULATES (does not provide) the independent
// strict-XSP/2 policy+process attestation and Nexus policy signing key.
package xmeshdrain

import (
    "encoding/hex"
    "encoding/json"
    "fmt"
    "os"
    "os/exec"
    "path/filepath"
    "strconv"
    "strings"
    "testing"
)

func TestV2SignedRealPostgreSQLTwoNodeProcess(t *testing.T) {
    signer:=os.Getenv("XMESH_CI_QUOTA_SQL_SIGNER")
    if signer=="" {t.Skip("only the disposable cross-repository PostgreSQL job")}
    if os.Getenv("GITHUB_ACTIONS")!="true" ||
        os.Getenv("RUNNER_ENVIRONMENT")!="github-hosted" ||
        os.Getenv("XMESH_AUDIT_DISPOSABLE")!="1" {
        t.Fatal("never request SQL-signed caps outside disposable hosted CI")
    }
    node,err:=strconv.ParseInt(os.Getenv("XMESH_CI_QUOTA_NODE"),10,64)
    if err!=nil || (node!=47 && node!=48){t.Fatal("unknown test Node")}
    dir,manager,p,_,_,_:=signedCapTestFixture(t)
    policyHash:=strings.Repeat("b",64)
    cmd:=exec.Command("python3",signer,"issue","--node",
        strconv.FormatInt(node,10),"--runtime",p.RuntimeID,
        "--policy-sha",policyHash,"--xray-pid",strconv.Itoa(os.Getpid()))
    output,err:=cmd.CombinedOutput()
    if err!=nil{t.Fatalf("real SQL preflight / signer failed: %s: %v",output,err)}
    var issued struct{
        NodeID int64 `json:"nodeId"`
        Ceiling uint64 `json:"ceiling"`
        SignedEnvelope string `json:"signedEnvelope"`
        PublicKeyHex string `json:"publicKeyHex"`
        Phase string `json:"phase"`
    }
    if err:=json.Unmarshal(output,&issued);err!=nil{
        t.Fatalf("bad independent SQL output: %v",err)
    }
    if issued.Phase!="sql-committed-signed-disposable" ||
        issued.NodeID!=node || issued.Ceiling==0 ||
        issued.Ceiling>100 {
        t.Fatal("SQL signer reported invalid grant")
    }
    pub,err:=hex.DecodeString(issued.PublicKeyHex)
    if err!=nil || len(pub)!=32{t.Fatal("invalid independent Nexus-style test key")}
    activation:="c"
    meter:="22222222-2222-4222-8222-222222222247"
    reservation:="33333333-3333-4333-8333-333333333347"
    if node==48 {
        activation="d"
        meter="22222222-2222-4222-8222-222222222248"
        reservation="33333333-3333-4333-8333-333333333348"
    }
    aliases:=[]signedCapAlias{
        {Email:capCDN,Class:"cdn"},
        {Email:capMain,Class:"main"},
    }
    context:=signedCapContext{
        NodeID:node,UserID:101,ActivationReceiptSHA256:strings.Repeat(activation,64),
        RuntimeID:p.RuntimeID,PolicySHA256:policyHash,
        AccountEpoch:"11111111-1111-4111-8111-111111111111",
        MeterEpoch:meter,ReservationID:reservation,
        ReservationRevision:1,
        InventoryTailDigest:strings.Repeat("0",64),
        Aliases:aliases,FirstGenesisIndependentlyProved:true,
        Current:func(alias,direction string)(uint64,error) {
            c:=manager.GetCounter(rawMeterName(alias,direction))
            if c==nil{return 0,nil}
            if c.Value()<0{return 0,fmt.Errorf("negative raw Xray bytes")}
            return uint64(c.Value()),nil
        },
    }
    gate.Lock()
    before:=checkSignedCapBeforeRegisterLocked(capMain)
    gate.Unlock()
    if before==nil{t.Fatal("unsigned Node was admitted before SQL reserve")}
    if err:=installSignedFirstGenesisCap(
        []byte(issued.SignedEnvelope),pub,context);err!=nil {
        t.Fatalf("verified SQL-backed Ed25519 grant rejected by Xray: %v",err)
    }
    cdnBytes:=int64(20)
    if issued.Ceiling<20 {cdnBytes=int64(issued.Ceiling)}
    mainBytes:=int64(issued.Ceiling)-cdnBytes
    cdn:=newCapCounter(t,manager,capCDN,"downlink")
    main:=newCapCounter(t,manager,capMain,"uplink")
    if !capAttempt(t,cdn,capCDN,"downlink",cdnBytes) {
        t.Fatal("validated per-Node CDN spending failed")
    }
    if mainBytes>0 && !capAttempt(t,main,capMain,"uplink",mainBytes){
        t.Fatal("validated remaining MAIN spending failed")
    }
    // All attempted extra bytes are denied *before* WAL mutation.
    if capAttempt(t,main,capMain,"uplink",1) ||
        capAttempt(t,cdn,capCDN,"downlink",1) {
        t.Fatal("one Node overspent its SQL-backed grant via another inbound")
    }
    all,cdnUsed:=capValues(t)
    if all!=issued.Ceiling || cdnUsed!=uint64(cdnBytes) {
        t.Fatalf("real SQL cap mismatch: %d/%d, expected %d/%d",
            all,cdnUsed,issued.Ceiling,cdnBytes)
    }
    if main.Value()!=mainBytes || cdn.Value()!=cdnBytes {
        t.Fatal("raw StatsService differs from accepted signed byte count")
    }
    mainTail,err:=readMeterTail(meterTestFile(t,dir,capMain,"uplink"))
    if err!=nil || mainTail.last.NextBytes!=uint64(mainBytes) {
        t.Fatalf("MAIN byte WAL lost committed bytes: %v",err)
    }
    cdnTail,err:=readMeterTail(meterTestFile(t,dir,capCDN,"downlink"))
    if err!=nil || cdnTail.last.NextBytes!=uint64(cdnBytes) {
        t.Fatalf("CDN byte WAL lost committed bytes: %v",err)
    }
    report:=struct {
        NodeID int64 `json:"nodeId"`
        GrantBytes uint64 `json:"grantBytes"`
        WalBytes uint64 `json:"walBytes"`
        RawBytes uint64 `json:"rawBytes"`
        RuntimeID string `json:"runtimeId"`
    }{node,issued.Ceiling,
        mainTail.last.NextBytes+cdnTail.last.NextBytes,
        uint64(main.Value()+cdn.Value()),p.RuntimeID}
    encoded,err:=json.Marshal(report)
    if err!=nil{t.Fatal(err)}
    dest:=filepath.Join(os.Getenv("XMESH_CI_QUOTA_RESULTS"),
        fmt.Sprintf("node-%d.json",node))
    if err:=os.WriteFile(dest,encoded,0600);err!=nil{t.Fatal(err)}
    t.Logf("PASS: measured Node %d spent exactly %d PostgreSQL reserved bytes; "+
        "no refunds or terminal metering claimed",node,issued.Ceiling)
}
