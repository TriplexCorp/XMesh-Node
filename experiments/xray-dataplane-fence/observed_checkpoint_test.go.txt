const testRevocation = "c5208c19-29bf-41b7-b90d-919574e01242"

func testScope(email,revocation string) checkpointScope {
    return checkpointScope{
        Email:email,NodeID:201,UserID:101,Revision:7,
        Epoch:"11111111-1111-4111-8111-111111111111",
        MeterEpoch:"22222222-2222-4222-8222-222222222222",
        ReservationID:"33333333-3333-4333-8333-333333333333",
        RevocationID:revocation,
    }
}

func setupCheckpoint(t *testing.T,email string,initialUp,initialDown int64) (string,func(int64,int64)) {
    t.Helper()
    resetGate()
    observed.Lock()
    observed.manager=nil
    observed.directory=""
    observed.runtimeID=""
    observed.Unlock()
    dir:=t.TempDir()
    // Testing.TempDir uses the process umask; journal requires explicit 0700.
    if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
    if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
    manager,err:=xstats.NewManager(context.Background(),&xstats.Config{})
    if err!=nil{t.Fatal(err)}
    BindStatsManager(manager)
    up,err:=manager.RegisterCounter("user>>>"+email+">>>traffic>>>uplink")
    if err!=nil{t.Fatal(err)}
    down,err:=manager.RegisterCounter("user>>>"+email+">>>traffic>>>downlink")
    if err!=nil{t.Fatal(err)}
    // Genuine initial usage precedes revocation; a post-fence increment is
    // now a violation, including if a later Set restores the same total.
    up.Add(initialUp)
    down.Add(initialDown)
    s:=ctx(666)
    if err:=Register(s,email,func(){});err!=nil{t.Fatal(err)}
    Finish(s)
    RouteFinish(s)
    fenceCtx,cancel:=context.WithCancel(context.Background())
    defer cancel()
    result,err:=fenceScoped(fenceCtx,testScope(email,testRevocation))
    if err!=nil||!result.Drained{t.Fatalf("not drained: %+v %v",result,err)}
    return dir,func(a,b int64){up.Add(a);down.Add(b)}
}

// Different StatsService counters are independent atomics, not a single
// transaction. A late async callback must invalidate a torn signed-scope
// observation before the immutable journal is published.
func TestObservedMeterTwoPassRejectsLateCounterGrowth(t *testing.T) {
    cdn:="xmesh-u101--vless-cdn"
    main:="xmesh-u101--vless-share"
    aliases:=[]string{cdn,main}
    values:=map[string]uint64{
        cdn+"/uplink":40,cdn+"/downlink":30,
        main+"/uplink":60,main+"/downlink":20,
    }
    calls:=0
    fixed:=func(alias,dir string)(uint64,error) {
        calls++
        return values[alias+"/"+dir],nil
    }
    rows,up,down,err:=readObservedMeterTwice(aliases,main,fixed)
    if err!=nil || up!=100 || down!=50 || len(rows)!=2 || calls!=8 {
        t.Fatalf("stable 2-inbound meter broken: %v %d %d %d %+v",err,up,down,calls,rows)
    }
    calls=0
    changing:=func(alias,dir string)(uint64,error) {
        calls++
        amount:=values[alias+"/"+dir]
        if calls>4 && alias==cdn && dir=="uplink" {amount++}
        return amount,nil
    }
    rows,up,down,err=readObservedMeterTwice(aliases,main,changing)
    if err==nil || rows!=nil || up!=0 || down!=0 {
        t.Fatalf("torn runtime counter silently captured: %v %+v %d %d",err,rows,up,down)
    }
    cases:=[]struct{name string;aliases []string;request string}{
        {"omitted_request",[]string{cdn},main},
        {"duplicate",[]string{cdn,cdn},main},
        {"unsorted",[]string{main,cdn},main},
        {"foreign_identity",[]string{cdn,"xmesh-u102--vless-share"},main},
    }
    for _,tc:=range cases {
        t.Run(tc.name,func(t *testing.T){
            if _,_,_,err:=readObservedMeterTwice(tc.aliases,tc.request,fixed);err==nil{
                t.Fatal("invalid meter identity list was accepted")
            }
        })
    }
    for _,mode:=range []string{"missing","overflow"} {
        t.Run(mode,func(t *testing.T) {
            read:=func(alias,dir string)(uint64,error){
                if mode=="missing" && alias==cdn && dir=="downlink" {
                    return 0,errors.New("missing CDN downlink")
                }
                if mode=="overflow" {return uint64(1<<63-1),nil}
                return values[alias+"/"+dir],nil
            }
            if _,_,_,err:=readObservedMeterTwice(aliases,main,read);err==nil {
                t.Fatalf("counter %s was accepted",mode)
            }
        })
    }
}

func TestObservedSignedMainAndCDNAliasesMustAggregateAllSeenTraffic(t *testing.T) {
    mainEmail:="xmesh-u101--vless-share"
    cdnEmail:="xmesh-u101--vless-cdn"
    for _,tc:=range []struct{
        name string
        missingCDN bool
        hiddenSameUser bool
    }{
        {name:"main-and-cdn"},
        {name:"missing-cdn-direction",missingCDN:true},
        {name:"hidden-same-user-pre-fence-traffic",hiddenSameUser:true},
    } {
        t.Run(tc.name,func(t *testing.T){
            resetGate()
            observed.Lock()
            observed.manager=nil
            observed.directory=""
            observed.runtimeID=""
            observed.Unlock()
            dir:=t.TempDir()
            if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
            if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
            manager,err:=xstats.NewManager(context.Background(),&xstats.Config{})
            if err!=nil{t.Fatal(err)}
            BindStatsManager(manager)
            addCounter:=func(email,direction string,n int64) {
                c,err:=manager.RegisterCounter("user>>>"+email+">>>traffic>>>"+direction)
                if err!=nil{t.Fatal(err)}
                c.Add(n)
            }
            addCounter(mainEmail,"uplink",100)
            addCounter(mainEmail,"downlink",60)
            addCounter(cdnEmail,"uplink",200)
            if !tc.missingCDN {addCounter(cdnEmail,"downlink",40)}
            if tc.hiddenSameUser {
                // A different provisioned inbound registered counters and
                // spent bytes BEFORE the fence, but its stream finished
                // before a Register() session was ever observed.
                hidden:="xmesh-u101--vless-hidden"
                addCounter(hidden,"uplink",37)
                addCounter(hidden,"downlink",19)
            }
            // A different billing user's counters MUST NOT contaminate A.
            addCounter("xmesh-u102--vless-other","uplink",5)
            addCounter("xmesh-u102--vless-other","downlink",7)
            p,c:=ctx(1011),ctx(1012)
            if err:=Register(p,mainEmail,func(){});err!=nil{t.Fatal(err)}
            if err:=Register(c,cdnEmail,func(){});err!=nil{t.Fatal(err)}
            Finish(p);RouteFinish(p)
            Finish(c);RouteFinish(c)
            scope:=testScope(mainEmail,testRevocation)
            snapshot,err:=fenceScoped(context.Background(),scope)
            if err!=nil||!snapshot.Drained{t.Fatalf("fence: %+v %v",snapshot,err)}
            observed,err:=persistObservedCheckpoint(scope)
            path:=checkpointPath(dir,mainEmail)
            if tc.missingCDN || tc.hiddenSameUser {
                if err==nil{t.Fatal("silently omitted a registered same-account meter")}
                if tc.hiddenSameUser && !strings.Contains(err.Error(),"unobserved same-user Xray StatsService counter") {
                    t.Fatalf("hidden provisioned alias rejected for wrong reason: %v",err)
                }
                if _,err:=os.Lstat(path);!errors.Is(err,os.ErrNotExist){
                    t.Fatalf("committed a partial multi-inbound observation: %v",err)
                }
                return
            }
            if err!=nil{t.Fatal(err)}
            if observed.UplinkBytes!=300 || observed.DownlinkBytes!=100 ||
               observed.FinalMeterVerified || observed.RefundableBytes!=0 {
                t.Fatalf("observed main/CDN bytes are not summed or were declared final: %+v",observed)
            }
            reread,err:=persistObservedCheckpoint(scope)
            if err!=nil||!reflect.DeepEqual(reread,observed) {
                t.Fatalf("multi-inbound replay changed immutable observation: %+v %v",reread,err)
            }
            // An alias may not mint a second per-user observed slot.
            if checkpointPath(dir,mainEmail)!=checkpointPath(dir,cdnEmail) {
                t.Fatal("alias bypassed immutable per-user observation")
            }
        })
    }
}

func TestZeroByteUnobservedAliasInvalidatesAlreadyPublishedNonfinalReplay(t *testing.T) {
    email:="xmesh-u101--vless-main"
    dir,_:=setupCheckpoint(t,email,30,40)
    scope:=testScope(email,testRevocation)
    first,err:=persistObservedCheckpoint(scope)
    if err!=nil{t.Fatal(err)}
    path:=checkpointPath(dir,email)
    original,err:=os.ReadFile(path)
    if err!=nil{t.Fatal(err)}
    // Even without a byte mutation, the counter registry proves that
    // the prior seen-session alias set was incomplete. It is not safe
    // to replay the older immutable observation as complete evidence.
    manager:=observed.manager
    for _,direction:=range []string{"uplink","downlink"} {
        name:="user>>>xmesh-u101--vless-hidden>>>traffic>>>"+direction
        if _,err:=manager.RegisterCounter(name);err!=nil{t.Fatal(err)}
    }
    if _,err:=persistObservedCheckpoint(scope);err==nil ||
        !strings.Contains(err.Error(),"unobserved same-user Xray StatsService counter") {
        t.Fatalf("zero-byte unexpected alias did not invalidate checkpoint: %v",err)
    }
    later,err:=os.ReadFile(path)
    if err!=nil{t.Fatal(err)}
    if !bytes.Equal(original,later) {
        t.Fatal("immutable observed journal changed after hidden alias detection")
    }
    if first.FinalMeterVerified || first.RefundableBytes!=0 {
        t.Fatal("stale observed snapshot incorrectly authorized a refund")
    }
}

func TestObservedFsyncSnapshotNeverRefundable(t *testing.T) {
    dir,add:=setupCheckpoint(t,"alice",100,230)
    record,err:=persistObservedCheckpoint(testScope("alice",testRevocation))
    if err!=nil{t.Fatal(err)}
    if record.SchemaVersion!=3||record.checkpointScope!=testScope("alice",testRevocation)||
       record.UplinkBytes!=100||record.DownlinkBytes!=230||!record.Drained||
       record.FinalMeterVerified||record.RefundableBytes!=0{
        t.Fatalf("invalid observed-only record: %+v",record)
    }
    file:=checkpointPath(dir,"alice")
    stat,err:=os.Lstat(file)
    if err!=nil{t.Fatal(err)}
    if stat.Mode().Perm()&0077!=0{t.Fatal("world-readable journal")}
    contents,err:=os.ReadFile(file)
    if err!=nil{t.Fatal(err)}
    repeat,err:=persistObservedCheckpoint(testScope("alice",testRevocation))
    if err!=nil||!reflect.DeepEqual(repeat,record){t.Fatalf("non-idempotent retry: %+v %v",repeat,err)}
    add(1,0)
    if _,err:=persistObservedCheckpoint(testScope("alice",testRevocation));err==nil{
        t.Fatal("late counter growth silently accepted immutable observation")
    }
    if _,err:=persistObservedCheckpoint(testScope("alice","d5208c19-29bf-41b7-b90d-919574e01242"));err==nil{
        t.Fatal("different revocation reused checkpoint")
    }
    if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
    if _,err:=persistObservedCheckpoint(testScope("alice",testRevocation));err==nil{
        t.Fatal("different Xray incarnation reused checkpoint")
    }
    after,err:=os.ReadFile(file)
    if err!=nil{t.Fatal(err)}
    if string(contents)!=string(after){t.Fatal("immutable observed file overwritten")}
}

// A readable, syntactically valid journal is NOT adequate for recovery.
// It must be the exact immutable bytes written by the observed-only recorder;
// duplicate fields and a false "final" claim can never become trustworthy.
func TestObservedCheckpointNoncanonicalRecoveryFailsClosed(t *testing.T) {
    email:="xmesh-u101--vless-main"
    for _,kind:=range []string{
        "duplicate-finality","duplicate-usage","unknown-proof",
        "trailing-blank","read-only-mode",
    } {
        t.Run(kind,func(t *testing.T){
            dir,_:=setupCheckpoint(t,email,11,13)
            scope:=testScope(email,testRevocation)
            if _,err:=persistObservedCheckpoint(scope);err!=nil{t.Fatal(err)}
            path:=checkpointPath(dir,email)
            b,err:=os.ReadFile(path)
            if err!=nil{t.Fatal(err)}
            data:=string(b)
            switch kind {
            case "duplicate-finality":
                data=strings.Replace(data,`"finalMeterVerified":false`,
                    `"finalMeterVerified":true,"finalMeterVerified":false`,1)
            case "duplicate-usage":
                data=strings.Replace(data,`"uplinkBytes":11`,
                    `"uplinkBytes":1,"uplinkBytes":11`,1)
            case "unknown-proof":
                data=strings.Replace(data,`"refundableBytes":0`,
                    `"refundableBytes":0,"terminalProof":"forged"`,1)
            case "trailing-blank":
                data+=" "
            case "read-only-mode":
                if err:=os.Chmod(path,0400);err!=nil{t.Fatal(err)}
            }
            if kind!="read-only-mode" {
                if data==string(b){t.Fatal("checkpoint tamper did not modify bytes")}
                if err:=os.WriteFile(path,[]byte(data),0600);err!=nil{t.Fatal(err)}
            }
            if _,err:=loadCheckpoint(path);err==nil {
                t.Fatal("ambiguous observed-only meter passed strict validation")
            }
            resetGate()
            if err:=configureObservedCheckpoint(dir);err==nil {
                t.Fatal("Xray restart accepted noncanonical observed meter")
            }
        })
    }
}

func TestConcurrentObservedRetryDoesNotReplace(t *testing.T) {
    dir,_:=setupCheckpoint(t,"dave",123,456)
    var wg sync.WaitGroup
    var mu sync.Mutex
    failed:=0
    for i:=0;i<12;i++ {
        wg.Add(1)
        go func(){
            defer wg.Done()
            record,err:=persistObservedCheckpoint(testScope("dave",testRevocation))
            if err!=nil || record.FinalMeterVerified || record.RefundableBytes!=0 {
                mu.Lock();failed++;mu.Unlock()
            }
        }()
    }
    wg.Wait()
    if failed!=0{t.Fatalf("concurrent idempotence failures: %d",failed)}
    if _,err:=os.Lstat(checkpointPath(dir,"dave"));err!=nil{t.Fatal(err)}
}

func TestPreFenceStatsServiceResetAndNegativeAddFailClosed(t *testing.T) {
    for _,kind:=range []string{"Set","NegativeAdd"}{
        for _,crashBeforeFence:=range []bool{false,true}{
            label:=kind
            if crashBeforeFence {label+="-crash-before-fence"}
            t.Run(label,func(t *testing.T){
                resetGate()
                observed.Lock()
                observed.manager=nil
                observed.directory=""
                observed.runtimeID=""
                observed.Unlock()
                dir:=t.TempDir()
                if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
                if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
                manager,err:=xstats.NewManager(context.Background(),&xstats.Config{})
                if err!=nil{t.Fatal(err)}
                BindStatsManager(manager)
                email:="xmesh-u101--vless-main"
                up,err:=manager.RegisterCounter("user>>>"+email+">>>traffic>>>uplink")
                if err!=nil{t.Fatal(err)}
                down,err:=manager.RegisterCounter("user>>>"+email+">>>traffic>>>downlink")
                if err!=nil{t.Fatal(err)}
                up.Add(100)
                down.Add(50)
                sessionCtx:=ctx(2201)
                if err:=Register(sessionCtx,email,func(){
                    Finish(sessionCtx)
                    RouteFinish(sessionCtx)
                });err!=nil{t.Fatal(err)}
                // No fence has been committed yet; even then, raw byte
                // rollback must produce a private durable taint FIRST.
                if kind=="Set"{up.Set(0)}else{up.Add(-7)}
                marker:=durableCounterTaintPath(dir,canonicalMeterUser(email))
                if _,err:=readDurableCounterTaint(marker);err!=nil{
                    t.Fatalf("pre-fence %s lost its fsynced taint: %v",kind,err)
                }
                if crashBeforeFence{
                    resetGate()
                    if err:=configureObservedCheckpoint(dir);err==nil{
                        t.Fatal("reset-before-fence orphan marker let Xray restart")
                    }
                    if _,err:=os.Lstat(marker);err!=nil{
                        t.Fatalf("unsafe crash evidence disappeared: %v",err)
                    }
                    return
                }
                // A legitimate later fence may stop this user, but it MUST
                // NEVER turn earlier erased bytes into a clean observation.
                scope:=testScope(email,testRevocation)
                stopped,err:=fenceScoped(context.Background(),scope)
                if err!=nil||!stopped.Drained{
                    t.Fatalf("fence after raw reset failed: %+v %v",stopped,err)
                }
                if _,err:=persistObservedCheckpoint(scope);err==nil{
                    t.Fatal("pre-fence reset yielded an accepted observed checkpoint")
                }
                if _,err:=os.Lstat(checkpointPath(dir,email));!errors.Is(err,os.ErrNotExist){
                    t.Fatalf("published false clean bytes after pre-fence reset: %v",err)
                }
                resetGate()
                if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
                recovered:=gate.users[canonicalMeterUser(email)]
                if recovered==nil || !recovered.fenced || !recovered.lateCounterMutation{
                    t.Fatal("pre-fence raw reset was forgotten after durable fence and restart")
                }
            })
        }
    }
}
func TestObservedLeaseScopeFailClosed(t *testing.T){
    dir,_:=setupCheckpoint(t,"scoped",0,0)
    valid:=testScope("scoped",testRevocation)
    if !validScope(valid) { t.Fatal("valid lease scope was rejected") }
    wrong:=[]checkpointScope{}
    corrupt:=func(f func(*checkpointScope)){ cp:=valid; f(&cp); wrong=append(wrong,cp) }
    corrupt(func(v *checkpointScope){v.NodeID=0})
    corrupt(func(v *checkpointScope){v.UserID=-1})
    corrupt(func(v *checkpointScope){v.Revision=0})
    corrupt(func(v *checkpointScope){v.Epoch="00000000-0000-0000-0000-000000000000"})
    corrupt(func(v *checkpointScope){v.MeterEpoch="broken"})
    corrupt(func(v *checkpointScope){v.ReservationID=""})
    corrupt(func(v *checkpointScope){v.RevocationID="broken"})
    corrupt(func(v *checkpointScope){v.Email=""})
    for _,cp:=range wrong {
        if validScope(cp) {t.Fatalf("accepted invalid scope: %+v",cp)}
        if _,err:=persistObservedCheckpoint(cp);err==nil {
            t.Fatalf("persisted invalid lease scope: %+v",cp)
        }
    }
    saved,err:=persistObservedCheckpoint(valid)
    if err!=nil {t.Fatal(err)}
    assertChanged:=func(name string, f func(*checkpointScope)){
        t.Helper()
        cp:=valid
        f(&cp)
        if _,err:=persistObservedCheckpoint(cp);err==nil{
            t.Fatalf("replayed checkpoint with substituted %s",name)
        }
        actual,err:=loadCheckpoint(checkpointPath(dir,valid.Email))
        if err!=nil || !reflect.DeepEqual(actual,saved) {
            t.Fatalf("original journal was changed: %+v %v",actual,err)
        }
    }
    assertChanged("node",func(x *checkpointScope){x.NodeID++})
    assertChanged("user",func(x *checkpointScope){x.UserID++})
    assertChanged("epoch",func(x *checkpointScope){x.Epoch="44444444-4444-4444-8444-444444444444"})
    assertChanged("meter",func(x *checkpointScope){x.MeterEpoch="55555555-5555-4555-8555-555555555555"})
    assertChanged("reservation",func(x *checkpointScope){x.ReservationID="66666666-6666-4666-8666-666666666666"})
    assertChanged("revision",func(x *checkpointScope){x.Revision++})
    assertChanged("revocation",func(x *checkpointScope){x.RevocationID="77777777-7777-4777-8777-777777777777"})
}

func TestObservedPublicationFailureNeverLosesNonfinalJournal(t *testing.T) {
    for _,mode:=range []string{"link-failure","directory-fsync-failure"} {
        t.Run(mode,func(t *testing.T){
            email:="observed-publish-"+mode
            dir,_:=setupCheckpoint(t,email,21,34)
            scope:=testScope(email,testRevocation)
            realLink,realSync:=experimentalJournalLink,experimentalJournalSync
            t.Cleanup(func(){experimentalJournalLink=realLink;experimentalJournalSync=realSync})
            if mode=="link-failure" {
                experimentalJournalLink=func(_, _ string) error {
                    return errors.New("injected observation hardlink failure")
                }
            }else{
                experimentalJournalSync=func(string) error {
                    return errors.New("injected observation directory fsync failure")
                }
            }
            _,err:=persistObservedCheckpoint(scope)
            experimentalJournalLink,experimentalJournalSync=realLink,realSync
            if err==nil{t.Fatal("uncommitted observation was acknowledged")}
            entries,err:=os.ReadDir(dir)
            if err!=nil{t.Fatal(err)}
            var pending string
            for _,entry:=range entries {
                if strings.HasPrefix(entry.Name(),".observed-pending-") {
                    if pending!=""{t.Fatal("multiple uncertain observation journals")}
                    pending=filepath.Join(dir,entry.Name())
                }
            }
            if pending==""{t.Fatal("uncertain observation was silently discarded")}
            final:=checkpointPath(dir,email)
            _,err=os.Lstat(final)
            if mode=="link-failure" {
                if !errors.Is(err,os.ErrNotExist){t.Fatalf("unexpected observed checkpoint: %v",err)}
                resetGate()
                if err:=configureObservedCheckpoint(dir);err==nil{
                    t.Fatal("unpublished observation allowed Xray restart")
                }
                if _,err:=os.Lstat(pending);err!=nil{
                    t.Fatalf("unpublished evidence was discarded: %v",err)
                }
            }else{
                if err!=nil{t.Fatal(err)}
                p,_:=os.Lstat(pending)
                f,_:=os.Lstat(final)
                if !os.SameFile(p,f){t.Fatal("post-link observation has a different inode")}
                resetGate()
                if err:=configureObservedCheckpoint(dir);err!=nil{
                    t.Fatalf("post-link observation was not recoverable: %v",err)
                }
                if _,err:=os.Lstat(pending);!errors.Is(err,os.ErrNotExist){
                    t.Fatalf("recovered observation pending file remains: %v",err)
                }
                recovered,err:=loadCheckpoint(final)
                if err!=nil || recovered.FinalMeterVerified || recovered.RefundableBytes!=0{
                    t.Fatalf("recovered record improperly claims finality: %+v %v",recovered,err)
                }
            }
        })
    }
}
