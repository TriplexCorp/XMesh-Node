const testRevocation = "c5208c19-29bf-41b7-b90d-919574e01242"

func testScope(email,revocation string) checkpointScope {
    return checkpointScope{
        Email:email,NodeID:201,UserID:101,Revision:7,
        Epoch:"11111111-1111-4111-8111-111111111111",
        MeterEpoch:"22222222-2222-4222-8222-222222222222",
        ReservationID:"33333333-3333-4333-8333-333333333333",
        RevocationID:revocation,
    }
}

func setupCheckpoint(t *testing.T,email string) (string,func(int64,int64)) {
    t.Helper()
    resetGate()
    observed.Lock()
    observed.manager=nil
    observed.directory=""
    observed.runtimeID=""
    observed.Unlock()
    dir:=t.TempDir()
    // Testing.TempDir uses the process umask; journal requires explicit 0700.
    if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
    if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
    manager,err:=xstats.NewManager(context.Background(),&xstats.Config{})
    if err!=nil{t.Fatal(err)}
    BindStatsManager(manager)
    up,err:=manager.RegisterCounter("user>>>"+email+">>>traffic>>>uplink")
    if err!=nil{t.Fatal(err)}
    down,err:=manager.RegisterCounter("user>>>"+email+">>>traffic>>>downlink")
    if err!=nil{t.Fatal(err)}
    s:=ctx(666)
    if err:=Register(s,email,func(){});err!=nil{t.Fatal(err)}
    Finish(s)
    RouteFinish(s)
    fenceCtx,cancel:=context.WithCancel(context.Background())
    defer cancel()
    result,err:=fenceScoped(fenceCtx,testScope(email,testRevocation))
    if err!=nil||!result.Drained{t.Fatalf("not drained: %+v %v",result,err)}
    return dir,func(a,b int64){up.Add(a);down.Add(b)}
}

func TestObservedSignedMainAndCDNAliasesMustAggregateAllSeenTraffic(t *testing.T) {
    mainEmail:="xmesh-u101--vless-share"
    cdnEmail:="xmesh-u101--vless-cdn"
    for _,missingCDN:=range []bool{false,true} {
        t.Run(fmt.Sprintf("missingCDNCounter=%v",missingCDN),func(t *testing.T){
            resetGate()
            observed.Lock()
            observed.manager=nil
            observed.directory=""
            observed.runtimeID=""
            observed.Unlock()
            dir:=t.TempDir()
            if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
            if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
            manager,err:=xstats.NewManager(context.Background(),&xstats.Config{})
            if err!=nil{t.Fatal(err)}
            BindStatsManager(manager)
            addCounter:=func(email,direction string,n int64) {
                c,err:=manager.RegisterCounter("user>>>"+email+">>>traffic>>>"+direction)
                if err!=nil{t.Fatal(err)}
                c.Add(n)
            }
            addCounter(mainEmail,"uplink",100)
            addCounter(mainEmail,"downlink",60)
            addCounter(cdnEmail,"uplink",200)
            if !missingCDN {addCounter(cdnEmail,"downlink",40)}
            p,c:=ctx(1011),ctx(1012)
            if err:=Register(p,mainEmail,func(){});err!=nil{t.Fatal(err)}
            if err:=Register(c,cdnEmail,func(){});err!=nil{t.Fatal(err)}
            Finish(p);RouteFinish(p)
            Finish(c);RouteFinish(c)
            scope:=testScope(mainEmail,testRevocation)
            snapshot,err:=fenceScoped(context.Background(),scope)
            if err!=nil||!snapshot.Drained{t.Fatalf("fence: %+v %v",snapshot,err)}
            observed,err:=persistObservedCheckpoint(scope)
            path:=checkpointPath(dir,mainEmail)
            if missingCDN {
                if err==nil{t.Fatal("silently omitted missing CDN traffic counter")}
                if _,err:=os.Lstat(path);!errors.Is(err,os.ErrNotExist){
                    t.Fatalf("committed a partial multi-inbound observation: %v",err)
                }
                return
            }
            if err!=nil{t.Fatal(err)}
            if observed.UplinkBytes!=300 || observed.DownlinkBytes!=100 ||
               observed.FinalMeterVerified || observed.RefundableBytes!=0 {
                t.Fatalf("observed main/CDN bytes are not summed or were declared final: %+v",observed)
            }
            reread,err:=persistObservedCheckpoint(scope)
            if err!=nil||reread!=observed {
                t.Fatalf("multi-inbound replay changed immutable observation: %+v %v",reread,err)
            }
            // An alias may not mint a second per-user observed slot.
            if checkpointPath(dir,mainEmail)!=checkpointPath(dir,cdnEmail) {
                t.Fatal("alias bypassed immutable per-user observation")
            }
        })
    }
}

func TestObservedFsyncSnapshotNeverRefundable(t *testing.T) {
    dir,add:=setupCheckpoint(t,"alice")
    add(100,230)
    record,err:=persistObservedCheckpoint(testScope("alice",testRevocation))
    if err!=nil{t.Fatal(err)}
    if record.SchemaVersion!=2||record.checkpointScope!=testScope("alice",testRevocation)||
       record.UplinkBytes!=100||record.DownlinkBytes!=230||!record.Drained||
       record.FinalMeterVerified||record.RefundableBytes!=0{
        t.Fatalf("invalid observed-only record: %+v",record)
    }
    file:=checkpointPath(dir,"alice")
    stat,err:=os.Lstat(file)
    if err!=nil{t.Fatal(err)}
    if stat.Mode().Perm()&0077!=0{t.Fatal("world-readable journal")}
    contents,err:=os.ReadFile(file)
    if err!=nil{t.Fatal(err)}
    repeat,err:=persistObservedCheckpoint(testScope("alice",testRevocation))
    if err!=nil||repeat!=record{t.Fatalf("non-idempotent retry: %+v %v",repeat,err)}
    add(1,0)
    if _,err:=persistObservedCheckpoint(testScope("alice",testRevocation));err==nil{
        t.Fatal("late counter growth silently accepted immutable observation")
    }
    if _,err:=persistObservedCheckpoint(testScope("alice","d5208c19-29bf-41b7-b90d-919574e01242"));err==nil{
        t.Fatal("different revocation reused checkpoint")
    }
    if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
    if _,err:=persistObservedCheckpoint(testScope("alice",testRevocation));err==nil{
        t.Fatal("different Xray incarnation reused checkpoint")
    }
    after,err:=os.ReadFile(file)
    if err!=nil{t.Fatal(err)}
    if string(contents)!=string(after){t.Fatal("immutable observed file overwritten")}
}

func TestObservedScopeMissingStatsAndInvalidIdentity(t *testing.T){
    dir,_:=setupCheckpoint(t,"bob")
    if _,err:=persistObservedCheckpoint(testScope("bob","bad-request"));err==nil{
        t.Fatal("accepted malformed revocation identity")
    }
    if _,err:=persistObservedCheckpoint(testScope("missing",testRevocation));err==nil{
        t.Fatal("accepted user with no drained session/counters")
    }
    if _,err:=os.Lstat(checkpointPath(dir,"missing"));!errors.Is(err,os.ErrNotExist){
        t.Fatalf("created unverifiable entry: %v",err)
    }
    unsafeDir:=filepath.Join(dir,"not-private")
    if err:=os.Mkdir(unsafeDir,0755);err!=nil{t.Fatal(err)}
    if err:=configureObservedCheckpoint(unsafeDir);err==nil{
        t.Fatal("accepted group-readable directory")
    }
}

func TestObservedSymlinkRefusedAndNoVictimOverwrite(t *testing.T){
    dir,add:=setupCheckpoint(t,"carol")
    add(5,10)
    victim:=filepath.Join(dir,"victim")
    if err:=os.WriteFile(victim,[]byte("UNCHANGED"),0600);err!=nil{t.Fatal(err)}
    if err:=os.Symlink(victim,checkpointPath(dir,"carol"));err!=nil{t.Fatal(err)}
    if _,err:=persistObservedCheckpoint(testScope("carol",testRevocation));err==nil{
        t.Fatal("existing symlink must fail closed")
    }
    b,err:=os.ReadFile(victim)
    if err!=nil||string(b)!="UNCHANGED"{t.Fatalf("victim mutated: %q %v",b,err)}
}

func TestConcurrentObservedRetryDoesNotReplace(t *testing.T) {
    dir,add:=setupCheckpoint(t,"dave")
    add(123,456)
    var wg sync.WaitGroup
    var mu sync.Mutex
    failed:=0
    for i:=0;i<12;i++ {
        wg.Add(1)
        go func(){
            defer wg.Done()
            record,err:=persistObservedCheckpoint(testScope("dave",testRevocation))
            if err!=nil || record.FinalMeterVerified || record.RefundableBytes!=0 {
                mu.Lock();failed++;mu.Unlock()
            }
        }()
    }
    wg.Wait()
    if failed!=0{t.Fatalf("concurrent idempotence failures: %d",failed)}
    if _,err:=os.Lstat(checkpointPath(dir,"dave"));err!=nil{t.Fatal(err)}
}

func TestObservedLeaseScopeFailClosed(t *testing.T){
    dir,_:=setupCheckpoint(t,"scoped")
    valid:=testScope("scoped",testRevocation)
    if !validScope(valid) { t.Fatal("valid lease scope was rejected") }
    wrong:=[]checkpointScope{}
    corrupt:=func(f func(*checkpointScope)){ cp:=valid; f(&cp); wrong=append(wrong,cp) }
    corrupt(func(v *checkpointScope){v.NodeID=0})
    corrupt(func(v *checkpointScope){v.UserID=-1})
    corrupt(func(v *checkpointScope){v.Revision=0})
    corrupt(func(v *checkpointScope){v.Epoch="00000000-0000-0000-0000-000000000000"})
    corrupt(func(v *checkpointScope){v.MeterEpoch="broken"})
    corrupt(func(v *checkpointScope){v.ReservationID=""})
    corrupt(func(v *checkpointScope){v.RevocationID="broken"})
    corrupt(func(v *checkpointScope){v.Email=""})
    for _,cp:=range wrong {
        if validScope(cp) {t.Fatalf("accepted invalid scope: %+v",cp)}
        if _,err:=persistObservedCheckpoint(cp);err==nil {
            t.Fatalf("persisted invalid lease scope: %+v",cp)
        }
    }
    saved,err:=persistObservedCheckpoint(valid)
    if err!=nil {t.Fatal(err)}
    assertChanged:=func(name string, f func(*checkpointScope)){
        t.Helper()
        cp:=valid
        f(&cp)
        if _,err:=persistObservedCheckpoint(cp);err==nil{
            t.Fatalf("replayed checkpoint with substituted %s",name)
        }
        actual,err:=loadCheckpoint(checkpointPath(dir,valid.Email))
        if err!=nil || actual!=saved {
            t.Fatalf("original journal was changed: %+v %v",actual,err)
        }
    }
    assertChanged("node",func(x *checkpointScope){x.NodeID++})
    assertChanged("user",func(x *checkpointScope){x.UserID++})
    assertChanged("epoch",func(x *checkpointScope){x.Epoch="44444444-4444-4444-8444-444444444444"})
    assertChanged("meter",func(x *checkpointScope){x.MeterEpoch="55555555-5555-4555-8555-555555555555"})
    assertChanged("reservation",func(x *checkpointScope){x.ReservationID="66666666-6666-4666-8666-666666666666"})
    assertChanged("revision",func(x *checkpointScope){x.Revision++})
    assertChanged("revocation",func(x *checkpointScope){x.RevocationID="77777777-7777-4777-8777-777777777777"})
}
