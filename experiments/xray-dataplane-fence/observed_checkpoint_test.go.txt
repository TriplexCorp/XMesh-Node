package xmeshdrain

import (
    "context"
    "errors"
    "os"
    "path/filepath"
    "sync"
    "testing"

    xstats "github.com/xtls/xray-core/app/stats"
)

const testRevocation = "c5208c19-29bf-41b7-b90d-919574e01242"

func setupCheckpoint(t *testing.T,email string) (string,func(int64,int64)) {
    t.Helper()
    resetGate()
    observed.Lock()
    observed.manager=nil
    observed.directory=""
    observed.runtimeID=""
    observed.Unlock()
    dir:=t.TempDir()
    // Testing.TempDir uses the process umask; journal requires explicit 0700.
    if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
    if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
    manager,err:=xstats.NewManager(context.Background(),&xstats.Config{})
    if err!=nil{t.Fatal(err)}
    BindStatsManager(manager)
    up,err:=manager.RegisterCounter("user>>>"+email+">>>traffic>>>uplink")
    if err!=nil{t.Fatal(err)}
    down,err:=manager.RegisterCounter("user>>>"+email+">>>traffic>>>downlink")
    if err!=nil{t.Fatal(err)}
    s:=ctx(666)
    if err:=Register(s,email,func(){});err!=nil{t.Fatal(err)}
    Finish(s)
    RouteFinish(s)
    fenceCtx,cancel:=context.WithCancel(context.Background())
    defer cancel()
    result,err:=fence(fenceCtx,email)
    if err!=nil||!result.Drained{t.Fatalf("not drained: %+v %v",result,err)}
    return dir,func(a,b int64){up.Add(a);down.Add(b)}
}

func TestObservedFsyncSnapshotNeverRefundable(t *testing.T) {
    dir,add:=setupCheckpoint(t,"alice")
    add(100,230)
    record,err:=persistObservedCheckpoint("alice",testRevocation)
    if err!=nil{t.Fatal(err)}
    if record.UplinkBytes!=100||record.DownlinkBytes!=230||!record.Drained||
       record.FinalMeterVerified||record.RefundableBytes!=0{
        t.Fatalf("invalid observed-only record: %+v",record)
    }
    file:=checkpointPath(dir,"alice")
    stat,err:=os.Lstat(file)
    if err!=nil{t.Fatal(err)}
    if stat.Mode().Perm()&0077!=0{t.Fatal("world-readable journal")}
    contents,err:=os.ReadFile(file)
    if err!=nil{t.Fatal(err)}
    repeat,err:=persistObservedCheckpoint("alice",testRevocation)
    if err!=nil||repeat!=record{t.Fatalf("non-idempotent retry: %+v %v",repeat,err)}
    add(1,0)
    if _,err:=persistObservedCheckpoint("alice",testRevocation);err==nil{
        t.Fatal("late counter growth silently accepted immutable observation")
    }
    if _,err:=persistObservedCheckpoint("alice","d5208c19-29bf-41b7-b90d-919574e01242");err==nil{
        t.Fatal("different revocation reused checkpoint")
    }
    if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
    if _,err:=persistObservedCheckpoint("alice",testRevocation);err==nil{
        t.Fatal("different Xray incarnation reused checkpoint")
    }
    after,err:=os.ReadFile(file)
    if err!=nil{t.Fatal(err)}
    if string(contents)!=string(after){t.Fatal("immutable observed file overwritten")}
}

func TestObservedScopeMissingStatsAndInvalidIdentity(t *testing.T){
    dir,_:=setupCheckpoint(t,"bob")
    if _,err:=persistObservedCheckpoint("bob","bad-request");err==nil{
        t.Fatal("accepted malformed revocation identity")
    }
    if _,err:=persistObservedCheckpoint("missing",testRevocation);err==nil{
        t.Fatal("accepted user with no drained session/counters")
    }
    if _,err:=os.Lstat(checkpointPath(dir,"missing"));!errors.Is(err,os.ErrNotExist){
        t.Fatalf("created unverifiable entry: %v",err)
    }
    unsafeDir:=filepath.Join(dir,"not-private")
    if err:=os.Mkdir(unsafeDir,0755);err!=nil{t.Fatal(err)}
    if err:=configureObservedCheckpoint(unsafeDir);err==nil{
        t.Fatal("accepted group-readable directory")
    }
}

func TestObservedSymlinkRefusedAndNoVictimOverwrite(t *testing.T){
    dir,add:=setupCheckpoint(t,"carol")
    add(5,10)
    victim:=filepath.Join(dir,"victim")
    if err:=os.WriteFile(victim,[]byte("UNCHANGED"),0600);err!=nil{t.Fatal(err)}
    if err:=os.Symlink(victim,checkpointPath(dir,"carol"));err!=nil{t.Fatal(err)}
    if _,err:=persistObservedCheckpoint("carol",testRevocation);err==nil{
        t.Fatal("existing symlink must fail closed")
    }
    b,err:=os.ReadFile(victim)
    if err!=nil||string(b)!="UNCHANGED"{t.Fatalf("victim mutated: %q %v",b,err)}
}

func TestConcurrentObservedRetryDoesNotReplace(t *testing.T) {
    dir,add:=setupCheckpoint(t,"dave")
    add(123,456)
    var wg sync.WaitGroup
    var mu sync.Mutex
    failed:=0
    for i:=0;i<12;i++ {
        wg.Add(1)
        go func(){
            defer wg.Done()
            record,err:=persistObservedCheckpoint("dave",testRevocation)
            if err!=nil || record.FinalMeterVerified || record.RefundableBytes!=0 {
                mu.Lock();failed++;mu.Unlock()
            }
        }()
    }
    wg.Wait()
    if failed!=0{t.Fatalf("concurrent idempotence failures: %d",failed)}
    if _,err:=os.Lstat(checkpointPath(dir,"dave"));err!=nil{t.Fatal(err)}
}
