func TestPostFenceRawCounterTaintSurvivesCrashRestoreAndABA(t *testing.T){
    email:="xmesh-u101--vless-main"
    dir,_:=setupCheckpoint(t,email,310,140)
    path:=durableCounterTaintPath(dir,canonicalMeterUser(email))
    c:=observed.manager.GetCounter("user>>>"+email+">>>traffic>>>uplink")
    before:=c.Value()
    c.Add(25)
    c.Set(before)
    marker,err:=readDurableCounterTaint(path)
    if err!=nil || marker.CanonicalUser!=canonicalMeterUser(email){
        t.Fatalf("late ABA was not durably tainted: %+v %v",marker,err)
    }
    if _,err=persistObservedCheckpoint(testScope(email,testRevocation));err==nil{
        t.Fatal("post-fence ABA was accepted before restart")
    }
    resetGate() // Simulate loss of every volatile audit flag.
    if err:=configureObservedCheckpoint(dir);err!=nil{t.Fatal(err)}
    manager,err:=xstats.NewManager(context.Background(),&xstats.Config{})
    if err!=nil{t.Fatal(err)}
    BindStatsManager(manager)
    restored:=gate.users[canonicalMeterUser(email)]
    if restored==nil || !restored.fenced || !restored.lateCounterMutation{
        t.Fatal("Xray lost durable post-fence discontinuity after restart")
    }
    if _,err=persistObservedCheckpoint(testScope(email,testRevocation));err==nil{
        t.Fatal("old raw counter ABA became an accepted checkpoint after restart")
    }
    if _,err=os.Lstat(checkpointPath(dir,email));!errors.Is(err,os.ErrNotExist){
        t.Fatal("recovery published a tainted observed meter")
    }
}

func TestCounterTaintPendingCrashRecovery(t *testing.T) {
    email:="xmesh-u101--vless-main"
    for _,kind:=range []string{"unpublished","linked","forged-inode","symlink","malformed"}{
        t.Run(kind,func(t *testing.T){
            dir,_:=setupCheckpoint(t,email,1,2)
            marker:=durableCounterTaint{SchemaVersion:1,CanonicalUser:canonicalMeterUser(email)}
            if err:=commitDurableCounterTaint(dir,marker.CanonicalUser);err!=nil{t.Fatal(err)}
            final:=durableCounterTaintPath(dir,marker.CanonicalUser)
            pending:=filepath.Join(dir,".taint-pending-test")
            switch kind{
            case "unpublished":
                if err:=os.Link(final,pending);err!=nil{t.Fatal(err)}
                if err:=os.Remove(final);err!=nil{t.Fatal(err)}
            case "linked":
                if err:=os.Link(final,pending);err!=nil{t.Fatal(err)}
            case "forged-inode":
                raw,err:=os.ReadFile(final)
                if err!=nil{t.Fatal(err)}
                if err:=os.WriteFile(pending,raw,0600);err!=nil{t.Fatal(err)}
            case "symlink":
                if err:=os.Symlink(final,pending);err!=nil{t.Fatal(err)}
            case "malformed":
                if err:=os.WriteFile(pending,[]byte(`{"schemaVersion":1,"canonicalUser":"xmesh-u101","canonicalUser":"xmesh-u101"}`+"\n"),0600);err!=nil{t.Fatal(err)}
            }
            resetGate()
            err:=configureObservedCheckpoint(dir)
            if kind=="linked" {
                if err!=nil{t.Fatalf("genuine post-link crash did not recover: %v",err)}
                if _,err:=os.Lstat(pending);!errors.Is(err,os.ErrNotExist){t.Fatalf("pending hardlink left behind: %v",err)}
                if !gate.users[canonicalMeterUser(email)].lateCounterMutation {
                    t.Fatal("recovered linked marker did not taint revoked user")
                }
                return
            }
            if err==nil{t.Fatalf("unsafe %s raw taint was silently accepted",kind)}
        })
    }
}

func TestCounterTaintRefusesOrphanAndUntrustedFile(t *testing.T) {
    for _,kind:=range []string{"orphan-fence","bad-identity","group-readable"}{
        t.Run(kind,func(t *testing.T){
            email:="xmesh-u101--vless-main"
            dir,_:=setupCheckpoint(t,email,5,7)
            path:=durableCounterTaintPath(dir,canonicalMeterUser(email))
            if err:=commitDurableCounterTaint(dir,canonicalMeterUser(email));err!=nil{t.Fatal(err)}
            switch kind {
            case "orphan-fence":
                if err:=os.Remove(durableFencePath(dir,email));err!=nil{t.Fatal(err)}
            case "bad-identity":
                raw,err:=os.ReadFile(path)
                if err!=nil{t.Fatal(err)}
                wrong:=append([]byte{},raw...)
                wrong[1]='X'
                if err:=os.WriteFile(path,wrong,0600);err!=nil{t.Fatal(err)}
            case "group-readable":
                if err:=os.Chmod(path,0644);err!=nil{t.Fatal(err)}
            }
            resetGate()
            if err:=configureObservedCheckpoint(dir);err==nil {
                t.Fatalf("unsafe %s taint was accepted at restart",kind)
            }
        })
    }
}

func TestRawCounterTaintPersistenceFailurePanicsBeforeMutation(t *testing.T) {
    email:="xmesh-u101--vless-main"
    dir,_:=setupCheckpoint(t,email,10,12)
    c:=observed.manager.GetCounter("user>>>"+email+">>>traffic>>>uplink")
    before:=c.Value()
    if err:=os.Chmod(dir,0755);err!=nil{t.Fatal(err)}
    defer os.Chmod(dir,0700)
    func(){
        defer func(){if recover()==nil {t.Error("unjournaled raw mutation did not fail closed")}}()
        c.Add(123)
    }()
    if c.Value()!=before{t.Fatal("unsafe raw counter was modified before durability")}
    if err:=os.Chmod(dir,0700);err!=nil{t.Fatal(err)}
    if _,err:=persistObservedCheckpoint(testScope(email,testRevocation));err==nil{
        t.Fatal("failed taint persistence left an unpoisoned volatile checkpoint")
    }
}

// A failed link or fsync may leave an unpublished raw StatsService taint.
// Restart must reject that evidence, or recover only the exact already
// published inode; no path may silently discard the pending file.
func TestRawCounterTaintPublicationFaultsRetainPendingEvidence(t *testing.T) {
    for _,mode:=range []string{"pending-fsync","link-failure","published-fsync"} {
        t.Run(mode,func(t *testing.T){
            email:="xmesh-u101--vless-main"
            dir,_:=setupCheckpoint(t,email,19,23)
            user:=canonicalMeterUser(email)
            realLink,realSync:=experimentalJournalLink,experimentalJournalSync
            t.Cleanup(func(){experimentalJournalLink=realLink;experimentalJournalSync=realSync})
            switch mode {
            case "pending-fsync":
                experimentalJournalSync=func(string) error {
                    return errors.New("injected pending taint dir fsync")
                }
            case "link-failure":
                experimentalJournalLink=func(_, _ string) error {
                    return errors.New("injected taint hardlink failure")
                }
            case "published-fsync":
                calls:=0
                experimentalJournalSync=func(directory string) error {
                    calls++
                    if calls==2 {return errors.New("injected final taint dir fsync")}
                    return realSync(directory)
                }
            }
            err:=commitDurableCounterTaint(dir,user)
            experimentalJournalLink,experimentalJournalSync=realLink,realSync
            if err==nil{t.Fatal("incomplete raw counter taint was acknowledged")}
            entries,err:=os.ReadDir(dir)
            if err!=nil{t.Fatal(err)}
            var pending string
            for _,entry:=range entries {
                if strings.HasPrefix(entry.Name(),".taint-pending-") {
                    if pending!="" {t.Fatal("multiple uncertain taint records")}
                    pending=filepath.Join(dir,entry.Name())
                }
            }
            if pending==""{t.Fatal("unsafe raw counter evidence silently discarded")}
            final:=durableCounterTaintPath(dir,user)
            if mode=="published-fsync" {
                p,err:=os.Lstat(pending)
                if err!=nil{t.Fatal(err)}
                f,err:=os.Lstat(final)
                if err!=nil{t.Fatal(err)}
                if !os.SameFile(p,f){t.Fatal("published pending taint has different inode")}
                resetGate()
                if err:=configureObservedCheckpoint(dir);err!=nil{
                    t.Fatalf("same-inode taint recovery failed: %v",err)
                }
                if _,err:=os.Lstat(pending);!errors.Is(err,os.ErrNotExist){
                    t.Fatalf("recovered pending taint remains: %v",err)
                }
                restored:=gate.users[user]
                if restored==nil || !restored.fenced || !restored.lateCounterMutation {
                    t.Fatal("restarted Xray dropped durable raw counter taint")
                }
            }else{
                if _,err:=os.Lstat(final);!errors.Is(err,os.ErrNotExist){
                    t.Fatalf("unpublished taint appeared committed: %v",err)
                }
                resetGate()
                if err:=configureObservedCheckpoint(dir);err==nil{
                    t.Fatal("unpublished taint allowed admission after restart")
                }
                if _,err:=os.Lstat(pending);err!=nil{
                    t.Fatalf("uncertain taint was deleted during restart: %v",err)
                }
            }
        })
    }
}

func TestRawCounterTaintLinkFailurePanicsBeforeByteMutation(t *testing.T) {
    email:="xmesh-u101--vless-main"
    dir,_:=setupCheckpoint(t,email,10,12)
    counter:=observed.manager.GetCounter("user>>>"+email+">>>traffic>>>uplink")
    before:=counter.Value()
    realLink:=experimentalJournalLink
    t.Cleanup(func(){experimentalJournalLink=realLink})
    experimentalJournalLink=func(_, _ string) error {
        return errors.New("injected hardlink failure")
    }
    func(){
        defer func(){
            if recover()==nil {t.Error("raw counter mutated despite failed durable taint")}
        }()
        counter.Add(123)
    }()
    experimentalJournalLink=realLink
    if counter.Value()!=before{t.Fatal("unsafe counter changed before taint durability")}
    pending,err:=filepath.Glob(filepath.Join(dir,".taint-pending-*"))
    if err!=nil || len(pending)!=1 {
        t.Fatalf("failed raw counter mutation lost its pending record: %v %+v",err,pending)
    }
    resetGate()
    if err:=configureObservedCheckpoint(dir);err==nil{
        t.Fatal("failed raw counter taint was forgotten on Xray restart")
    }
}
