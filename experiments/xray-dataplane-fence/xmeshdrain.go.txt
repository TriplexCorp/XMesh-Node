// Package xmeshdrain is a CI-ONLY experiment injected into an exact pinned
// upstream Xray source tree. It does not verify leases, persist final meters,
// create signed evidence, or authorize any refundable bytes.

var errFenced = errors.New("experimental xmesh drain: user is fenced")
var errNoSessionID = errors.New("experimental xmesh drain: missing session ID")
var errNoUserID = errors.New("experimental xmesh drain: missing authenticated user email")
var errDuplicateSession = errors.New("experimental xmesh drain: duplicate live session ID")
var enabled = os.Getenv("GITHUB_ACTIONS") == "true" &&
    os.Getenv("XMESH_EXPERIMENTAL_DRAIN") == "1"

// Every user-bearing inbound must be enumerated. A single-inbound fence
// cannot protect a second CDN/secondary listener carrying the same identity.
var trackedInboundTags = map[string]struct{}{"vless-share": {}}
var signedUserEmail = regexp.MustCompile(`^xmesh-u([1-9][0-9]{0,18})--([^\x00-\x20>]{1,128})$`)

// A signed policy emits distinct per-inbound emails for the same billing
// user. Always fence the stable account identity, not the individual tag.
// Unknown experimental identities remain distinct; no loose prefix merging.
func canonicalMeterUser(email string) string {
    if matches := signedUserEmail.FindStringSubmatch(email); matches != nil {
        return "xmesh-u" + matches[1]
    }
    return email
}

func parseTrackedInboundTags(raw string) (map[string]struct{}, error) {
    if raw == "" {
        return map[string]struct{}{"vless-share": {}}, nil
    }
    parts := strings.Split(raw, ",")
    if len(parts) == 0 || len(parts) > 16 {
        return nil, errors.New("invalid experimental inbound coverage count")
    }
    allowed := make(map[string]struct{}, len(parts))
    for _, tag := range parts {
        if len(tag) == 0 || len(tag) > 64 {
            return nil, errors.New("empty or oversized tracked inbound")
        }
        for _, c := range tag {
            if !((c >= 'a' && c <= 'z') ||
                 (c >= 'A' && c <= 'Z') ||
                 (c >= '0' && c <= '9') || c == '-' || c == '_') {
                return nil, errors.New("unsafe experimental inbound tag")
            }
        }
        if _, exists := allowed[tag]; exists {
            return nil, errors.New("duplicate tracked inbound tag")
        }
        allowed[tag] = struct{}{}
    }
    return allowed, nil
}

// Reject a blank email on an explicitly covered inbound, not just nonblank
// emails. Otherwise an authenticated connection could skip Register.
func IsTrackedInbound(ctx context.Context) bool {
    if !enabled { return false }
    subject := session.InboundFromContext(ctx)
    if subject == nil || subject.User == nil { return false }
    _, ok := trackedInboundTags[subject.Tag]
    return ok
}

// Every authenticated user stream is billable on this disposable Xray.
// Missing a configured inbound must not silently provide an unmetered path.
// Never interpret coverage success as a proof of terminal dataplane bytes.
func RequireInboundCoverage(ctx context.Context) error {
    if !enabled { return nil }
    subject := session.InboundFromContext(ctx)
    if subject == nil { return nil }
    // A private StatsService call does not need user authentication.
    if subject.Tag == "xmesh_stats_in" { return nil }
    // Never let a tagged dataplane route proceed without user attribution,
    // even if upstream Xray omitted the authenticated User context. Checking
    // only for User != nil permits silent unmetered traffic on that route.
    if subject.Tag != "" && (subject.User == nil || subject.User.Email == "") {
        return errors.New("experimental Xray: unattributed user-bearing inbound")
    }
    if subject.User != nil && !IsTrackedInbound(ctx) {
        return errors.New("experimental Xray: authenticated user on untracked inbound")
    }
    return nil
}

type trackedSession struct {
    email string
    stop func()
    // Process() and the route's outbound task are independent lifetimes.
    workerFinished bool
    routeFinished bool
}
type userState struct {
    fenced bool
    active map[uint64]func()
    // Capture all authenticated per-inbound Xray emails seen for this stable
    // billing user. This set is NOT a proof of complete signed topology.
    emails map[string]struct{}
    // Count all in-flight per-user metered writes. New writes are rejected
    // atomically once the admission fence is active.
    writes int
    // Per exact inbound email + direction; unrelated active traffic cannot
    // authorize a direct StatsService mutation or wrong-direction write.
    meterOps map[string]int
    // Track asynchronous TimeoutWrapperReader operations through their
    // counter increment, including callbacks resumed AFTER a timeout.
    reads int
    // Transport cancellation runs in its own goroutine. A completed inbound
    // worker/route is NOT enough to claim drained if its close callback has
    // not yet returned (for example, a blocked network/socket Close).
    stopping int
    // Any direct user-byte Counter.Add/Set after fencing poisons the meter.
    lateCounterMutation bool
    done chan struct{}
    closed bool
}
var gate = struct {
    sync.Mutex
    users map[string]*userState
    sessions map[uint64]*trackedSession
    // Root-private, fsynced journal selected before the Xray listener starts.
    journalDir string
}{users: map[string]*userState{}, sessions: map[uint64]*trackedSession{}}

func ensureUser(email string) *userState {
    u := gate.users[email]
    if u == nil {
        u = &userState{active: make(map[uint64]func()), emails: make(map[string]struct{}), meterOps: make(map[string]int), done: make(chan struct{})}
        gate.users[email] = u
    }
    return u
}

// Register is called during authenticated Dispatch/DispatchLink BEFORE
// returning a transport link. The worker must call Finish after Process ends.
func Register(ctx context.Context, email string, stop func()) error {
    if !enabled {
        return nil
    }
    // An authenticated connection with no scoped user identity must not
    // silently bypass tracking: a later drain would otherwise be incomplete.
    if email == "" {
        return errNoUserID
    }
    id := uint64(c.IDFromContext(ctx))
    if id == 0 {
        return errNoSessionID
    }
    if stop == nil {
        return errors.New("experimental xmesh drain: missing transport cancellation")
    }
    gate.Lock()
    defer gate.Unlock()
    u := ensureUser(canonicalMeterUser(email))
    if u.fenced {
        return errFenced
    }
    if existing := gate.sessions[id]; existing != nil {
        if existing.email != canonicalMeterUser(email) {
            return errors.New("experimental xmesh drain: session ID changed user")
        }
        // Reusing a live ID could leave an untracked transport open after
        // Finish(id), falsely reporting a complete drain. Refuse it even
        // when the email matches; the caller closes the rejected stream.
        return errDuplicateSession
    }
    u.active[id] = stop
    u.emails[email] = struct{}{}
    gate.sessions[id] = &trackedSession{email: canonicalMeterUser(email), stop: stop}
    return nil
}

// ScopedEmail returns the billable identity of every explicitly covered
// disposable inbound. Never affect the management or unrelated listeners.
func ScopedEmail(ctx context.Context) string {
    if !IsTrackedInbound(ctx) { return "" }
    return session.InboundFromContext(ctx).User.Email
}

// RequireMeteringCoverage fails closed BEFORE registering an authenticated
// billable stream. Xray normally skips SizeStatWriter/TimeoutWrapperReader
// counters if a user's policy disables either direction, or if stats counter
// registration fails. Admitting that stream would allow real unmetered bytes
// even though a later observed-only checkpoint might reject missing counters.
//
// Caller is already inside RequireInboundCoverage + IsTrackedInbound and must
// close both transport links if validation fails. The counter lookup itself
// is from Xray's real StatsService, not a caller-supplied claimed boolean.
func RequireMeteringCoverage(uplinkEnabled, downlinkEnabled bool, counterReady func(string) bool) error {
    if !enabled { return nil }
    if !uplinkEnabled || !downlinkEnabled || counterReady == nil {
        return errors.New("experimental Xray: mandatory user uplink/downlink stats disabled")
    }
    for _, direction := range []string{"uplink", "downlink"} {
        if !counterReady(direction) {
            return fmt.Errorf("experimental Xray: missing mandatory %s user meter", direction)
        }
    }
    return nil
}

// Positive user-byte mutations MUST originate from the explicit metered
// counter method AND match the exact in-flight inbound alias+direction.
// A direct Counter.Add() is never authorized merely because another goroutine
// is currently writing that same alias. The trusted provenance bit is set
// only inside the experimentally instrumented StatsService metered method. An out-of-band counter Add (even before revocation), a
// raw Set/negative Add, and a post-fence mutation all poison future settlement
// BEFORE the value changes. This is a crash-durable NONFINAL taint, not proof
// that all Xray transports or kernel queues have stopped.
func rawMeterName(email,direction string) string {
    if email=="" || (direction!="uplink" && direction!="downlink") {
        panic("experimental xmesh drain: invalid metered alias or direction")
    }
    return "user>>>"+email+">>>traffic>>>"+direction
}
func meterDirection(directions []string, fallback string) string {
    if len(directions)>1 {panic("experimental xmesh drain: multiple meter directions")}
    if len(directions)==1 {return directions[0]}
    return fallback // Only legacy unit-fixture callers omit an explicit direction.
}
func auditRawCounter(name string, preFenceUnsafe bool, trustedMeterOrigin bool, overflow func() bool, mutate func() int64) int64 {
    if !enabled {return mutate()}
    raw,ok:=strings.CutPrefix(name,"user>>>")
    if !ok {return mutate()}
    email,direction,ok:=strings.Cut(raw,">>>traffic>>>")
    if !ok || email=="" || (direction!="uplink" && direction!="downlink") {
        return mutate()
    }
    gate.Lock()
    defer gate.Unlock()
    u:=ensureUser(canonicalMeterUser(email))
    exceeds:=overflow!=nil && overflow()
    if u.fenced || preFenceUnsafe || exceeds || !trustedMeterOrigin || u.meterOps[name]<=0 {
        u.lateCounterMutation=true
        if err:=commitDurableCounterTaint(gate.journalDir,canonicalMeterUser(email));err!=nil {
            panic(fmt.Sprintf("experimental Xray counter taint cannot be persisted; deny: %v",err))
        }
    }
    if exceeds {
        // int64 wraps from positive to negative; a second wrap may even
        // appear positive again. Never permit the first ambiguous mutation.
        panic("experimental Xray raw user byte counter overflow denied")
    }
    return mutate()
}
func AuditRawCounterMutation(name string, preFenceUnsafe bool, mutate func() int64) int64 {
    return auditRawCounter(name,preFenceUnsafe,false,nil,mutate)
}
// AuditRawCounterAdd checks the old signed 64-bit StatsService byte value and
// the new delta under the SAME gate as fence and checkpoint publication.
func AuditRawCounterAdd(name string,delta int64,trustedMeterOrigin bool,current func() int64,mutate func() int64) int64 {
    if delta==0 {return mutate()}
    return auditRawCounter(name,delta<0,trustedMeterOrigin,func() bool {
        if delta<=0 {return false}
        n:=current()
        return n<0 || n>int64(1<<63-1)-delta
    },mutate)
}

// BeginMeterWrite covers the entire billable SizeStatWriter.WriteMultiBuffer.
// No new starts after revocation; the meter alias MUST match the exact raw
// counter being incremented by this tracked transport write.
func BeginMeterWrite(email string,directions ...string) error {
    if !enabled || email=="" {return nil}
    name:=rawMeterName(email,meterDirection(directions,"downlink"))
    gate.Lock()
    defer gate.Unlock()
    u:=ensureUser(canonicalMeterUser(email))
    if u.fenced {return errFenced}
    u.writes++
    u.meterOps[name]++
    return nil
}
func EndMeterWrite(email string,directions ...string) {
    if !enabled || email=="" {return}
    name:=rawMeterName(email,meterDirection(directions,"downlink"))
    gate.Lock()
    defer gate.Unlock()
    u:=ensureUser(canonicalMeterUser(email))
    if u.writes<=0 || u.meterOps[name]<=0 {
        panic("experimental xmesh drain: unbalanced exact-direction metered write")
    }
    u.meterOps[name]--
    u.writes--
    completeUserIfIdleLocked(u)
}

// Every reader counted by WrapLink belongs to the uplink direction. Tracking
// the exact alias prevents a concurrent CDN uplink read from "authorizing"
// an unexplained MAIN uplink counter mutation.
func BeginMeterRead(email string) bool {
    if !enabled || email=="" {return true}
    name:=rawMeterName(email,"uplink")
    gate.Lock()
    defer gate.Unlock()
    u:=ensureUser(canonicalMeterUser(email))
    if u.fenced {return false}
    u.reads++
    u.meterOps[name]++
    return true
}
func EndMeterRead(email string) {
    if !enabled || email=="" {return}
    name:=rawMeterName(email,"uplink")
    gate.Lock()
    defer gate.Unlock()
    u:=ensureUser(canonicalMeterUser(email))
    if u.reads<=0 || u.meterOps[name]<=0 {
        panic("experimental xmesh drain: unbalanced exact-alias metered read")
    }
    u.meterOps[name]--
    u.reads--
    completeUserIfIdleLocked(u)
}

// MayDeliverMeterRead is checked AFTER a timed-out asynchronous read has
// completed and its raw bytes were accounted. A transport can be fenced while
// the read is blocked; returning its old buffer after the fence would make it
// available to a detached outbound writer. It is intentionally separate from
// BeginMeterRead: existing in-flight reads must still finish accounting, while
// their buffers are discarded if the user's admission has since been revoked.
// This check is NOT an atomic authorization for downstream delivery.
func MayDeliverMeterRead(email string) bool {
    if !enabled || email=="" {return false}
    gate.Lock()
    defer gate.Unlock()
    u:=gate.users[canonicalMeterUser(email)]
    // A pre-fence counter discontinuity poisons settlement/observation,
// but does not itself revoke an unrelated, still-authorized live stream.
    return u!=nil && !u.fenced
}

// Caller must hold gate.Mutex. Do not report a drained user while any
// logical session, metered operation, or transport cancellation is unfinished.
func completeUserIfIdleLocked(u *userState) {
    if u.fenced && len(u.active) == 0 && u.writes == 0 && u.reads == 0 && u.stopping == 0 && !u.closed {
        close(u.done)
        u.closed = true
    }
}

// Finish is run by the inbound worker AFTER the authenticated proxy's
// Process returned and the transport has been closed. This is a best-effort
// experimental boundary, NOT a verified byte-accounting flush.
func Finish(ctx context.Context) {
    if !enabled {
        return
    }
    id := uint64(c.IDFromContext(ctx))
    if id == 0 {
        return
    }
    gate.Lock()
    defer gate.Unlock()
    session := gate.sessions[id]
    if session == nil { return }
    session.workerFinished = true
    completeIfFinishedLocked(id,session)
}

// RouteFinish runs after the asynchronous routedDispatch() returns.
// This strengthens the stop barrier but is NOT final accounting proof:
// Xray outbound implementations may launch further background writers.
func RouteFinish(ctx context.Context) {
    if !enabled { return }
    id := uint64(c.IDFromContext(ctx))
    if id == 0 { return }
    gate.Lock()
    defer gate.Unlock()
    session := gate.sessions[id]
    if session == nil { return }
    session.routeFinished = true
    completeIfFinishedLocked(id,session)
}

// Caller holds gate.Mutex. Do not complete on worker exit alone.
func completeIfFinishedLocked(id uint64,session *trackedSession) {
    if !session.workerFinished || !session.routeFinished { return }
    delete(gate.sessions,id)
    u := gate.users[session.email]
    delete(u.active,id)
    completeUserIfIdleLocked(u)
}

type result struct {
    Email string `json:"email"`
    Fenced bool `json:"fenced"`
    Active int `json:"active"`
    Drained bool `json:"drained"`
    FinalMeterVerified bool `json:"finalMeterVerified"`
    Refundable bool `json:"refundable"`
    // This record is fsynced but explicitly NOT authoritative final traffic.
    ObservedCheckpoint *observedCheckpoint `json:"observedCheckpoint,omitempty"`
}

func state(email string, u *userState) result {
    return result{
        Email: email, Fenced: u.fenced, Active: len(u.active),
        Drained: u.closed && len(u.active) == 0 && u.writes == 0 && u.reads == 0 && u.stopping == 0,
        FinalMeterVerified: false, Refundable: false,
    }
}

func fence(ctx context.Context, email string) (result, error) {
    gate.Lock()
    u := ensureUser(canonicalMeterUser(email))
    var cancels []func()
    if !u.fenced {
        u.fenced = true
        for _, closeConn := range u.active {
            cancels = append(cancels, closeConn)
        }
        // Register all cancellation tasks BEFORE checking completion or
        // releasing the gate. Otherwise the last Finish/RouteFinish can close
        // done while a transport close is still outstanding.
        u.stopping += len(cancels)
        completeUserIfIdleLocked(u)
    }
    done := u.done
    gate.Unlock()
    for _, closeConn := range cancels {
        go func(stop func()) {
            defer func() {
                gate.Lock()
                u.stopping--
                if u.stopping < 0 {
                    gate.Unlock()
                    panic("experimental xmesh drain: unbalanced transport cancellation")
                }
                completeUserIfIdleLocked(u)
                gate.Unlock()
            }()
            stop()
        }(closeConn)
    }
    select {
    case <-ctx.Done():
        return result{}, ctx.Err()
    case <-done:
    }
    gate.Lock()
    defer gate.Unlock()
    return state(email, u), nil
}

// A control-socket request is NOT cryptographically authenticated Node lease
// evidence. Even on the root-only socket, do not let a duplicate JSON key
// silently rebind the requested account or revocation. Encoding/json's normal
// struct Decode accepts duplicate keys, and the previous handler accepted a
// second JSON object after the first. Both are ambiguous for a stop intent.
func decodeScopedFenceRequest(body io.Reader) (checkpointScope,error) {
    var req checkpointScope
    raw,err:=io.ReadAll(io.LimitReader(body,2049))
    if err!=nil{return req,err}
    if len(raw)>2048{return req,errors.New("oversized scoped fence request")}
    d:=json.NewDecoder(bytes.NewReader(raw))
    first,err:=d.Token()
    if err!=nil || first!=json.Delim('{') {
        return req,errors.New("scoped fence request must be one JSON object")
    }
    keys:=make(map[string]struct{},8)
    for d.More() {
        token,err:=d.Token()
        if err!=nil{return req,err}
        key,ok:=token.(string)
        if !ok{return req,errors.New("invalid scoped fence field")}
        if _,duplicate:=keys[key];duplicate {
            return req,errors.New("duplicate scoped fence field")
        }
        keys[key]=struct{}{}
        var value json.RawMessage
        if err:=d.Decode(&value);err!=nil{return req,err}
    }
    closing,err:=d.Token()
    if err!=nil || closing!=json.Delim('}') {
        return req,errors.New("unterminated scoped fence object")
    }
    var trailing json.RawMessage
    if err:=d.Decode(&trailing);!errors.Is(err,io.EOF) {
        return req,errors.New("trailing scoped fence JSON")
    }
    strict:=json.NewDecoder(bytes.NewReader(raw))
    strict.DisallowUnknownFields()
    if err:=strict.Decode(&req);err!=nil{return req,err}
    if !validScope(req){return req,errors.New("invalid scoped lease identity")}
    return req,nil
}

func serveHTTP(w http.ResponseWriter, r *http.Request) {
    w.Header().Set("Content-Type", "application/json")
    if r.Method != http.MethodPost || r.URL.Path != "/v1/fence" {
        http.Error(w, "not found", http.StatusNotFound)
        return
    }
    // This is accessible only through a root-owned local Unix socket
    // in a disposable CI runner. It is NEVER a production Node control API.
    req,err:=decodeScopedFenceRequest(r.Body)
    if err!=nil {
        http.Error(w, "invalid scoped request", http.StatusBadRequest)
        return
    }
    ctx, cancel := context.WithTimeout(r.Context(), 6*time.Second)
    defer cancel()
    snapshot, err := fenceScoped(ctx, req)
    if err != nil {
        code:=http.StatusConflict
        if errors.Is(err,context.DeadlineExceeded)||errors.Is(err,context.Canceled) {
            code=http.StatusRequestTimeout
        }
        http.Error(w, fmt.Sprintf("incomplete drain: %v", err),code)
        return
    }
    // Controlled SIGKILL only in disposable CI: prove fsynced deny intent
    // survives a crash BEFORE any observed counter snapshot is committed.
    if os.Getenv("XMESH_EXPERIMENTAL_CRASH_AFTER_DRAIN")=="1" {
        _ = syscall.Kill(os.Getpid(),syscall.SIGKILL)
        panic("SIGKILL did not stop disposable Xray")
    }
    observedRecord, err := persistObservedCheckpoint(req)
    if err != nil {
        // No trustworthy checkpoint was committed; do NOT issue a success
        // response even if the user's TCP sessions were physically stopped.
        http.Error(w, "observed checkpoint rejected: "+err.Error(),http.StatusConflict)
        return
    }
    snapshot.ObservedCheckpoint = &observedRecord
    _ = json.NewEncoder(w).Encode(snapshot)
}

func init() {
    if !enabled {
        return
    }
    coverage, err := parseTrackedInboundTags(
        os.Getenv("XMESH_EXPERIMENTAL_SCOPED_INBOUNDS"))
    if err != nil { panic(err) }
    trackedInboundTags = coverage
    // This experiment never runs without a durable, private *observed*
    // journal. The file still does not authorize refunds.
    dir:=os.Getenv("XMESH_EXPERIMENTAL_OBSERVED_DIR")
    if err:=lockExperimentalInstance(dir);err!=nil {panic(err)}
    if err := configureObservedCheckpoint(dir); err != nil {
        panic(err)
    }
    socket := os.Getenv("XMESH_EXPERIMENTAL_DRAIN_SOCKET")
    if socket == "" {
        panic("experimental xmesh drain socket required")
    }
    parent, err := os.Stat(filepath.Dir(socket))
    if err != nil || !parent.IsDir() || parent.Mode().Perm()&0077 != 0 {
        panic("experimental xmesh drain parent must exist with private permissions")
    }
    if err:=recoverStaleUnixSocket(socket);err!=nil{panic(err)}
    listener, err := net.Listen("unix", socket)
    if err != nil {
        panic(err)
    }
    if err := os.Chmod(socket, 0600); err != nil {
        _ = listener.Close()
        panic(err)
    }
    go func() {
        _ = (&http.Server{Handler: http.HandlerFunc(serveHTTP),
            ReadHeaderTimeout: 2*time.Second}).Serve(listener)
    }()
}
