// Package xmeshdrain is a CI-ONLY experiment injected into an exact pinned
// upstream Xray source tree. It does not verify leases, persist final meters,
// create signed evidence, or authorize any refundable bytes.
package xmeshdrain

import (
    "context"
    "encoding/json"
    "errors"
    "fmt"
    "net"
    "net/http"
    "os"
    "path/filepath"
    "sync"
    "time"

    c "github.com/xtls/xray-core/common/ctx"
)

var errFenced = errors.New("experimental xmesh drain: user is fenced")
var errNoSessionID = errors.New("experimental xmesh drain: missing session ID")
var errNoUserID = errors.New("experimental xmesh drain: missing authenticated user email")
var errDuplicateSession = errors.New("experimental xmesh drain: duplicate live session ID")
var enabled = os.Getenv("GITHUB_ACTIONS") == "true" &&
    os.Getenv("XMESH_EXPERIMENTAL_DRAIN") == "1"

type trackedSession struct {
    email string
    stop func()
}
type userState struct {
    fenced bool
    active map[uint64]func()
    done chan struct{}
    closed bool
}
var gate = struct {
    sync.Mutex
    users map[string]*userState
    sessions map[uint64]*trackedSession
}{users: map[string]*userState{}, sessions: map[uint64]*trackedSession{}}

func ensureUser(email string) *userState {
    u := gate.users[email]
    if u == nil {
        u = &userState{active: make(map[uint64]func()), done: make(chan struct{})}
        gate.users[email] = u
    }
    return u
}

// Register is called during authenticated Dispatch/DispatchLink BEFORE
// returning a transport link. The worker must call Finish after Process ends.
func Register(ctx context.Context, email string, stop func()) error {
    if !enabled {
        return nil
    }
    // An authenticated connection with no scoped user identity must not
    // silently bypass tracking: a later drain would otherwise be incomplete.
    if email == "" {
        return errNoUserID
    }
    id := uint64(c.IDFromContext(ctx))
    if id == 0 {
        return errNoSessionID
    }
    if stop == nil {
        return errors.New("experimental xmesh drain: missing transport cancellation")
    }
    gate.Lock()
    defer gate.Unlock()
    u := ensureUser(email)
    if u.fenced {
        return errFenced
    }
    if existing := gate.sessions[id]; existing != nil {
        if existing.email != email {
            return errors.New("experimental xmesh drain: session ID changed user")
        }
        // Reusing a live ID could leave an untracked transport open after
        // Finish(id), falsely reporting a complete drain. Refuse it even
        // when the email matches; the caller closes the rejected stream.
        return errDuplicateSession
    }
    u.active[id] = stop
    gate.sessions[id] = &trackedSession{email: email, stop: stop}
    return nil
}

// Finish is run by the inbound worker AFTER the authenticated proxy's
// Process returned and the transport has been closed. This is a best-effort
// experimental boundary, NOT a verified byte-accounting flush.
func Finish(ctx context.Context) {
    if !enabled {
        return
    }
    id := uint64(c.IDFromContext(ctx))
    if id == 0 {
        return
    }
    gate.Lock()
    defer gate.Unlock()
    session := gate.sessions[id]
    if session == nil {
        return
    }
    delete(gate.sessions, id)
    u := gate.users[session.email]
    delete(u.active, id)
    if u.fenced && len(u.active) == 0 && !u.closed {
        close(u.done)
        u.closed = true
    }
}

type result struct {
    Email string `json:"email"`
    Fenced bool `json:"fenced"`
    Active int `json:"active"`
    Drained bool `json:"drained"`
    FinalMeterVerified bool `json:"finalMeterVerified"`
    Refundable bool `json:"refundable"`
}

func state(email string, u *userState) result {
    return result{
        Email: email, Fenced: u.fenced, Active: len(u.active),
        Drained: u.closed && len(u.active) == 0,
        FinalMeterVerified: false, Refundable: false,
    }
}

func fence(ctx context.Context, email string) (result, error) {
    gate.Lock()
    u := ensureUser(email)
    var cancels []func()
    if !u.fenced {
        u.fenced = true
        for _, closeConn := range u.active {
            cancels = append(cancels, closeConn)
        }
        if len(u.active) == 0 && !u.closed {
            close(u.done)
            u.closed = true
        }
    }
    done := u.done
    gate.Unlock()
    for _, closeConn := range cancels {
        go closeConn()
    }
    select {
    case <-ctx.Done():
        return result{}, ctx.Err()
    case <-done:
    }
    gate.Lock()
    defer gate.Unlock()
    return state(email, u), nil
}

func serveHTTP(w http.ResponseWriter, r *http.Request) {
    w.Header().Set("Content-Type", "application/json")
    if r.Method != http.MethodPost || r.URL.Path != "/v1/fence" {
        http.Error(w, "not found", http.StatusNotFound)
        return
    }
    // This is accessible only through a root-owned local Unix socket
    // in a disposable CI runner. It is NEVER a production Node control API.
    var req struct { Email string `json:"email"` }
    r.Body = http.MaxBytesReader(w, r.Body, 512)
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil ||
        len(req.Email) == 0 || len(req.Email) > 128 {
        http.Error(w, "invalid email", http.StatusBadRequest)
        return
    }
    ctx, cancel := context.WithTimeout(r.Context(), 6*time.Second)
    defer cancel()
    snapshot, err := fence(ctx, req.Email)
    if err != nil {
        http.Error(w, fmt.Sprintf("incomplete drain: %v", err),
            http.StatusRequestTimeout)
        return
    }
    _ = json.NewEncoder(w).Encode(snapshot)
}

func init() {
    if !enabled {
        return
    }
    socket := os.Getenv("XMESH_EXPERIMENTAL_DRAIN_SOCKET")
    if socket == "" {
        panic("experimental xmesh drain socket required")
    }
    parent, err := os.Stat(filepath.Dir(socket))
    if err != nil || !parent.IsDir() || parent.Mode().Perm()&0077 != 0 {
        panic("experimental xmesh drain parent must exist with private permissions")
    }
    if _, err := os.Lstat(socket); err == nil {
        panic("experimental xmesh drain refuses preexisting socket/symlink")
    } else if !os.IsNotExist(err) {
        panic(err)
    }
    listener, err := net.Listen("unix", socket)
    if err != nil {
        panic(err)
    }
    if err := os.Chmod(socket, 0600); err != nil {
        _ = listener.Close()
        panic(err)
    }
    go func() {
        _ = (&http.Server{Handler: http.HandlerFunc(serveHTTP),
            ReadHeaderTimeout: 2*time.Second}).Serve(listener)
    }()
}
