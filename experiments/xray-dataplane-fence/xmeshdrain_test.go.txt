func TestScopedFenceControlRejectsDuplicateAndTrailingJSON(t *testing.T) {
    valid:=testScope("xmesh-u101--vless-main",testRevocation)
    canonical,err:=json.Marshal(valid)
    if err!=nil{t.Fatal(err)}
    for _,raw:=range []string{
        string(canonical),
        " \n" + string(canonical) + "\n",
    } {
        parsed,err:=decodeScopedFenceRequest(strings.NewReader(raw))
        if err!=nil || parsed!=valid {
            t.Fatalf("valid scoped request failed: %v %+v",err,parsed)
        }
    }
    for _,tc:=range []struct{name string;raw string}{
        {"duplicate_user_id",strings.Replace(string(canonical),
            `"userId":101`,`"userId":102,"userId":101`,1)},
        {"duplicate_email",strings.Replace(string(canonical),
            `"email":"xmesh-u101--vless-main"`,
            `"email":"xmesh-u102--vless-main","email":"xmesh-u101--vless-main"`,1)},
        {"duplicate_revocation",strings.Replace(string(canonical),
            `"revocationId":"`+testRevocation+`"`,
            `"revocationId":"44444444-4444-4444-8444-444444444444","revocationId":"`+testRevocation+`"`,1)},
        {"unknown_terminal_claim",strings.Replace(string(canonical),
            `"revision":7`,`"revision":7,"finalMeterVerified":true`,1)},
        {"trailing_json",string(canonical)+"{}"},
        {"trailing_nonjson",string(canonical)+" invalid"},
        {"array", "["+string(canonical)+"]"},
        {"oversized",string(canonical)+strings.Repeat(" ",2050)},
        {"mismatched_billing_user",strings.Replace(string(canonical),
            `"userId":101`,`"userId":102`,1)},
        {"noncanonical_email",strings.Replace(string(canonical),
            `"email":"xmesh-u101--vless-main"`,`"email":"xmesh-u0101--vless-main"`,1)},
    } {
        t.Run(tc.name,func(t *testing.T){
            if bytes.Equal([]byte(tc.raw),canonical){t.Fatal("test failed to tamper request")}
            if _,err:=decodeScopedFenceRequest(strings.NewReader(tc.raw));err==nil{
                t.Fatal("ambiguous, forged or out-of-scope fence request admitted")
            }
        })
    }
}

func TestMultiInboundCoverageIsStrictAndIncludesBlankUserEmails(t *testing.T) {
    original := trackedInboundTags
    defer func() { trackedInboundTags = original }()
    tags, err := parseTrackedInboundTags("vless-share,vless-cdn")
    if err != nil || len(tags) != 2 { t.Fatalf("coverage: %v %v", tags, err) }
    trackedInboundTags = tags
    scoped := func(tag, email string) context.Context {
        return session.ContextWithInbound(context.Background(), &session.Inbound{
            Tag: tag, User: &protocol.MemoryUser{Email: email},
        })
    }
    for _, tag := range []string{"vless-share", "vless-cdn"} {
        scopedCtx := scoped(tag, "same-account")
        if !IsTrackedInbound(scopedCtx) || ScopedEmail(scopedCtx) != "same-account" {
            t.Fatalf("billable inbound %s bypassed session accounting", tag)
        }
        blank := scoped(tag, "")
        if !IsTrackedInbound(blank) || ScopedEmail(blank) != "" {
            t.Fatal("blank authenticated email bypasses Register error path")
        }
    }
    if IsTrackedInbound(scoped("xmesh_stats_in", "same-account")) ||
        ScopedEmail(scoped("xmesh_stats_in", "same-account")) != "" {
        t.Fatal("private management StatsService was classified as a user stream")
    }
    if err := RequireInboundCoverage(scoped("xmesh_stats_in", "management")); err != nil {
        t.Fatalf("private StatsService must remain accessible: %v", err)
    }
    if err := RequireInboundCoverage(scoped("vless-cdn", "same-account")); err != nil {
        t.Fatalf("explicit secondary inbound unexpectedly refused: %v", err)
    }
    if err := RequireInboundCoverage(scoped("vless-cdn", "")); err == nil {
        t.Fatal("tagged user inbound with blank email skipped the coverage gate")
    }
    noUser := session.ContextWithInbound(context.Background(), &session.Inbound{
        Tag: "vless-cdn",
    })
    if err := RequireInboundCoverage(noUser); err == nil {
        t.Fatal("tagged inbound with missing authenticated User was unmetered")
    }
    if err := RequireInboundCoverage(scoped("vless-unlisted", "")); err == nil {
        t.Fatal("unknown user-bearing inbound with blank email was unmetered")
    }
    if err := RequireInboundCoverage(session.ContextWithInbound(
        context.Background(), &session.Inbound{Tag: "xmesh_stats_in"})); err != nil {
        t.Fatalf("private StatsService without User was blocked: %v",err)
    }
    if err := RequireInboundCoverage(scoped("vless-unlisted", "same-account")); err == nil {
        t.Fatal("an untracked user-bearing inbound bypassed quota accounting")
    }
    if err := RequireInboundCoverage(context.Background()); err != nil {
        t.Fatalf("no authenticated session should not be rejected: %v", err)
    }
    if IsTrackedInbound(context.Background()) {
        t.Fatal("unattributed context was treated as a tracked session")
    }
    for _, raw := range []string{
        ",vless-share", "vless-share,", "vless-share,vless-share",
        "vless-share, vless-cdn", "vless-share,xmesh_stats_in!",
        "vless-share/other", "vless-share\n", "vless-share,vless-cdn,",
    } {
        if _, err := parseTrackedInboundTags(raw); err == nil {
            t.Fatalf("invalid coverage list accepted: %q", raw)
        }
    }
    fallback, err := parseTrackedInboundTags("")
    if err != nil || len(fallback) != 1 {
        t.Fatal("legacy isolated test inbound was not preserved")
    }
    if _, ok := fallback["vless-share"]; !ok {
        t.Fatal("default test inbound missing")
    }
}

// A counted user may not be admitted when either stats direction was
// disabled by Xray policy, or when StatsService cannot create a counter.
// Unit fixtures exercise the strict gate; pinned-real-Xray CI additionally
// checks the dispatcher patch on actual authenticated VLESS streams.
func TestRequireMeteringCoverageBeforeAdmission(t *testing.T) {
    tested := []string{}
    present := func(direction string) bool {
        tested=append(tested,direction)
        return direction=="uplink" || direction=="downlink"
    }
    if err:=RequireMeteringCoverage(true,true,present);err!=nil{
        t.Fatalf("valid bidirectional stats denied: %v",err)
    }
    if len(tested)!=2 || tested[0]!="uplink" || tested[1]!="downlink" {
        t.Fatalf("must require BOTH directions: %v",tested)
    }
    for _,tc:=range []struct{
        name string
        up,down bool
        ready func(string) bool
    }{
        {"no-uplink",false,true,present},
        {"no-downlink",true,false,present},
        {"neither",false,false,present},
        {"nil-counter-lookup",true,true,nil},
        {"failed-uplink-counter",true,true,func(s string) bool {return s!="uplink"}},
        {"failed-downlink-counter",true,true,func(s string) bool {return s!="downlink"}},
    } {
        t.Run(tc.name,func(t *testing.T){
            if err:=RequireMeteringCoverage(tc.up,tc.down,tc.ready);err==nil {
                t.Fatal("authenticated user with missing byte counter was admitted")
            }
        })
    }
}

func TestSignedUserIdentityAcrossMainAndCDNEmailsCannotBypassFence(t *testing.T) {
    resetGate()
    mainEmail := "xmesh-u101--vless-main"
    cdnEmail := "xmesh-u101--vless-cdn"
    if canonicalMeterUser(mainEmail) != canonicalMeterUser(cdnEmail) ||
        canonicalMeterUser(mainEmail) != "xmesh-u101" {
        t.Fatal("signed same-user accounting identities not combined")
    }
    for _, tag := range []string{"xmesh-u101--PL%2FCDN", "xmesh-u101--🇵🇱CDN"} {
        if canonicalMeterUser(tag) != "xmesh-u101" {
            t.Fatalf("signed encoded transport tag escaped user fence: %s",tag)
        }
    }
    for _, malformed := range []string{"xmesh-u01--vless-main", "xmesh-u0--cdn",
        "xmesh-u101--", "xmesh-u101--invalid>tag", "xmesh-u102--cdn"} {
        if canonicalMeterUser(malformed) == canonicalMeterUser(mainEmail) {
            t.Fatalf("an unrelated or invalid identity was merged: %s",malformed)
        }
    }
    primary, secondary := ctx(2801),ctx(2802)
    stopped := make(chan string, 2)
    if err := Register(primary,mainEmail,func(){
        Finish(primary)
        RouteFinish(primary)
        stopped <- "main"
    }); err != nil { t.Fatal(err) }
    if err := Register(secondary,cdnEmail,func(){
        Finish(secondary)
        RouteFinish(secondary)
        stopped <- "cdn"
    }); err != nil { t.Fatal(err) }
    if err := BeginMeterWrite(cdnEmail); err != nil { t.Fatal(err) }
    outcome := make(chan result,1)
    failure := make(chan error,1)
    go func(){
        deadline,cancel := context.WithTimeout(context.Background(),2*time.Second)
        defer cancel()
        value,err := fence(deadline,mainEmail)
        if err != nil { failure <- err; return }
        outcome <- value
    }()
    <-stopped
    <-stopped
    for _, email := range []string{mainEmail,cdnEmail} {
        if err := Register(ctx(2803),email,func(){}); !errors.Is(err,errFenced) {
            t.Fatalf("revoked tag %s admitted another stream: %v",email,err)
        }
        if err := BeginMeterWrite(email); !errors.Is(err,errFenced) {
            t.Fatalf("revoked tag %s admitted another billable write: %v",email,err)
        }
        if BeginMeterRead(email) {
            t.Fatalf("revoked tag %s admitted another async meter read",email)
        }
    }
    select {
    case <-outcome: t.Fatal("fence completed before CDN writer finished")
    case err := <-failure: t.Fatalf("premature error: %v",err)
    case <-time.After(35*time.Millisecond):
    }
    EndMeterWrite(cdnEmail)
    select {
    case value := <-outcome:
        if !value.Fenced || !value.Drained || value.Active!=0 ||
           value.FinalMeterVerified || value.Refundable {
            t.Fatalf("incorrect canonical-user fence: %+v",value)
        }
    case err := <-failure:
        t.Fatal(err)
    case <-time.After(time.Second):
        t.Fatal("canonical-user fence did not finish")
    }
}

// A transport cancellation may finish its worker and route callbacks before
// the underlying Close returns. We must not claim a completed stop during that
// gap, even if the last logical session and all meter calls are gone.
func TestFenceWaitsForTransportCancellationCallback(t *testing.T) {
    resetGate()
    sessionCtx:=ctx(921)
    stopStarted:=make(chan struct{})
    releaseStop:=make(chan struct{})
    var releaseOnce sync.Once
    unblock:=func(){releaseOnce.Do(func(){close(releaseStop)})}
    defer unblock()
    if err:=Register(sessionCtx,"slow-close",func(){
        Finish(sessionCtx)
        RouteFinish(sessionCtx)
        close(stopStarted)
        <-releaseStop
    });err!=nil{t.Fatal(err)}
    type response struct { snapshot result; err error }
    completed:=make(chan response,1)
    deadline,cancel:=context.WithTimeout(context.Background(),150*time.Millisecond)
    defer cancel()
    go func(){
        snapshot,err:=fence(deadline,"slow-close")
        completed<-response{snapshot,err}
    }()
    select {
    case <-stopStarted:
    case <-time.After(2*time.Second):
        t.Fatal("stop callback did not execute")
    }
    gate.Lock()
    u:=gate.users["slow-close"]
    active,stopping,closed:=len(u.active),u.stopping,u.closed
    gate.Unlock()
    if active!=0 || stopping!=1 || closed {
        t.Fatalf("stop barrier ignored unfinished callback: active=%d stopping=%d closed=%v",
            active,stopping,closed)
    }
    select {
    case got:=<-completed:
        if !errors.Is(got.err,context.DeadlineExceeded) || got.snapshot.Drained ||
            got.snapshot.FinalMeterVerified || got.snapshot.Refundable {
            t.Fatalf("unfinished close produced acknowledged drain: %+v",got)
        }
    case <-time.After(2*time.Second):
        t.Fatal("unfinished callback did not time out")
    }
    if err:=Register(ctx(922),"slow-close",func(){});!errors.Is(err,errFenced) {
        t.Fatalf("timeout reopened user admission: %v",err)
    }
    unblock()
    finalCtx,finalCancel:=context.WithTimeout(context.Background(),2*time.Second)
    defer finalCancel()
    done,err:=fence(finalCtx,"slow-close")
    if err!=nil || !done.Drained || done.Active!=0 ||
        done.FinalMeterVerified || done.Refundable {
        t.Fatalf("callback completion was not idempotently recovered: %+v %v",done,err)
    }
    gate.Lock()
    remaining:=gate.users["slow-close"].stopping
    gate.Unlock()
    if remaining!=0 {t.Fatalf("stop callbacks leaked: %d",remaining)}
}

func TestManyConcurrentRegistrationsCannotCrossFence(t *testing.T) {
    resetGate()
    const count=180
    var wg sync.WaitGroup
    var mu sync.Mutex
    ids:=make([]uint32,0,count)
    for i:=0;i<count;i++ {
        wg.Add(1)
        go func(i int) {
            defer wg.Done()
            id:=uint32(i+100)
            c:=ctx(id)
            if err:=Register(c,"crowd",func(){Finish(c);RouteFinish(c)}); err==nil {
                mu.Lock()
                ids=append(ids,id)
                mu.Unlock()
            } else if !errors.Is(err,errFenced) {
                t.Errorf("invalid admission failure: %v",err)
            }
        }(i)
    }
    stopper := make(chan struct{})
    go func() {
        deadline,cancel:=context.WithTimeout(context.Background(),3*time.Second)
        defer cancel()
        info,err:=fence(deadline,"crowd")
        if err!=nil || !info.Fenced || !info.Drained || info.Active!=0 ||
            info.Refundable || info.FinalMeterVerified {
            t.Errorf("incorrect concurrent fence result: %+v %v",info,err)
        }
        close(stopper)
    }()
    wg.Wait()
    <-stopper
    mu.Lock()
    admitted:=len(ids)
    mu.Unlock()
    if admitted>count { t.Fatalf("admission count exceeded: %d",admitted) }
    if err:=Register(ctx(300),"crowd",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("concurrent re-entry: %v",err)
    }
}

// A pending read that began before revocation must still be accounted, but
// it must NEVER deliver the buffered result to an outbound after revocation.
func TestMeterReadDeliveryAuthorizationRevokedAcrossInboundAliases(t *testing.T){
    resetGate()
    mainEmail:="xmesh-u101--vless-main"
    cdnEmail:="xmesh-u101--vless-cdn"
    unrelated:="xmesh-u102--vless-main"
    a:=ctx(901)
    b:=ctx(902)
    stopped:=make(chan struct{})
    if err:=Register(a,mainEmail,func(){close(stopped)});err!=nil{t.Fatal(err)}
    if err:=Register(b,unrelated,func(){});err!=nil{t.Fatal(err)}
    if !BeginMeterRead(cdnEmail) || !MayDeliverMeterRead(mainEmail) ||
        !MayDeliverMeterRead(cdnEmail) {
        t.Fatal("healthy signed alias read was unexpectedly denied")
    }
    if !MayDeliverMeterRead(unrelated){t.Fatal("independent live user denied")}
    // A StatsService reset before B's revocation invalidates B's meter,
    // but is not itself a transport-denial decision. Keep its live stream
    // usable until its own signed revocation arrives.
    gate.Lock()
    gate.users[canonicalMeterUser(unrelated)].lateCounterMutation=true
    gate.Unlock()
    if !MayDeliverMeterRead(unrelated){
        t.Fatal("pre-fence taint must not prematurely stop user B")
    }
    outcome:=make(chan struct{r result;err error},1)
    go func(){
        ctx,cancel:=context.WithTimeout(context.Background(),time.Second)
        defer cancel()
        r,err:=fence(ctx,mainEmail)
        outcome<-struct{r result;err error}{r,err}
    }()
    <-stopped
    Finish(a)
    RouteFinish(a)
    if MayDeliverMeterRead(mainEmail)||MayDeliverMeterRead(cdnEmail){
        t.Fatal("fenced main/CDN account may still deliver a pending buffer")
    }
    if !MayDeliverMeterRead(unrelated){t.Fatal("fence denied unrelated user")}
    if BeginMeterRead(cdnEmail){t.Fatal("revocation admitted a new CDN read")}
    select{
    case got:=<-outcome:
        t.Fatalf("pending read falsely acknowledged drained: %+v",got)
    case <-time.After(25*time.Millisecond):
    }
    EndMeterRead(cdnEmail)
    select{
    case got:=<-outcome:
        if got.err!=nil||!got.r.Drained||got.r.FinalMeterVerified||got.r.Refundable{
            t.Fatalf("read completion incorrectly created terminal claim: %+v",got)
        }
    case <-time.After(time.Second):
        t.Fatal("drain not completed after last read")
    }
    Finish(b)
    RouteFinish(b)
    if MayDeliverMeterRead(mainEmail)||MayDeliverMeterRead(cdnEmail){
        t.Fatal("drained signed user reopened buffered delivery")
    }
}

func TestMeterReadBarrierIncludesTimedOutAsyncReader(t *testing.T){
    resetGate()
    stream:=ctx(302)
    stopped:=make(chan struct{})
    if err:=Register(stream,"pending-read",func(){close(stopped)});err!=nil{
        t.Fatal(err)
    }
    if !BeginMeterRead("pending-read"){t.Fatal("failed to start test read")}
    outcome:=make(chan struct{info result;err error},1)
    go func(){
        bound,cancel:=context.WithTimeout(context.Background(),time.Second)
        defer cancel()
        info,err:=fence(bound,"pending-read")
        outcome<-struct{info result;err error}{info,err}
    }()
    <-stopped
    Finish(stream)
    RouteFinish(stream)
    if BeginMeterRead("pending-read"){
        t.Fatal("revocation allowed a new metered read")
    }
    select {
    case got:=<-outcome:
        t.Fatalf("premature drain while timed-out read pending: %+v",got)
    case <-time.After(50*time.Millisecond):
    }
    EndMeterRead("pending-read")
    select {
    case got:=<-outcome:
        if got.err!=nil||!got.info.Drained||got.info.FinalMeterVerified||
            got.info.Refundable {
            t.Fatalf("completion falsely authorizes refundable bytes: %+v",got)
        }
    case <-time.After(2*time.Second):
        t.Fatal("drain never completed after pending read was accounted")
    }
    if BeginMeterRead("pending-read"){
        t.Fatal("read admission reopened after drain")
    }
}
