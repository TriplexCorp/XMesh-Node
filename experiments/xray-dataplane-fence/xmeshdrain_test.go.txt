func TestRuntimeFingerprintStableForSameLiveKernelProcess(t *testing.T) {
    first,err:=currentRuntimeIdentity()
    if err!=nil{t.Fatal(err)}
    second,err:=currentRuntimeIdentity()
    if err!=nil{t.Fatal(err)}
    if len(first)!=64 || first!=second {
        t.Fatalf("Linux process identity changed between same-process reads: %q %q",first,second)
    }
}

func TestMultiInboundCoverageIsStrictAndIncludesBlankUserEmails(t *testing.T) {
    original := trackedInboundTags
    defer func() { trackedInboundTags = original }()
    tags, err := parseTrackedInboundTags("vless-share,vless-cdn")
    if err != nil || len(tags) != 2 { t.Fatalf("coverage: %v %v", tags, err) }
    trackedInboundTags = tags
    scoped := func(tag, email string) context.Context {
        return session.ContextWithInbound(context.Background(), &session.Inbound{
            Tag: tag, User: &protocol.MemoryUser{Email: email},
        })
    }
    for _, tag := range []string{"vless-share", "vless-cdn"} {
        scopedCtx := scoped(tag, "same-account")
        if !IsTrackedInbound(scopedCtx) || ScopedEmail(scopedCtx) != "same-account" {
            t.Fatalf("billable inbound %s bypassed session accounting", tag)
        }
        blank := scoped(tag, "")
        if !IsTrackedInbound(blank) || ScopedEmail(blank) != "" {
            t.Fatal("blank authenticated email bypasses Register error path")
        }
    }
    if IsTrackedInbound(scoped("xmesh_stats_in", "same-account")) ||
        ScopedEmail(scoped("xmesh_stats_in", "same-account")) != "" {
        t.Fatal("private management StatsService was classified as a user stream")
    }
    if err := RequireInboundCoverage(scoped("xmesh_stats_in", "management")); err != nil {
        t.Fatalf("private StatsService must remain accessible: %v", err)
    }
    if err := RequireInboundCoverage(scoped("vless-cdn", "same-account")); err != nil {
        t.Fatalf("explicit secondary inbound unexpectedly refused: %v", err)
    }
    if err := RequireInboundCoverage(scoped("vless-cdn", "")); err == nil {
        t.Fatal("tagged user inbound with blank email skipped the coverage gate")
    }
    noUser := session.ContextWithInbound(context.Background(), &session.Inbound{
        Tag: "vless-cdn",
    })
    if err := RequireInboundCoverage(noUser); err == nil {
        t.Fatal("tagged inbound with missing authenticated User was unmetered")
    }
    if err := RequireInboundCoverage(scoped("vless-unlisted", "")); err == nil {
        t.Fatal("unknown user-bearing inbound with blank email was unmetered")
    }
    if err := RequireInboundCoverage(session.ContextWithInbound(
        context.Background(), &session.Inbound{Tag: "xmesh_stats_in"})); err != nil {
        t.Fatalf("private StatsService without User was blocked: %v",err)
    }
    if err := RequireInboundCoverage(scoped("vless-unlisted", "same-account")); err == nil {
        t.Fatal("an untracked user-bearing inbound bypassed quota accounting")
    }
    if err := RequireInboundCoverage(context.Background()); err != nil {
        t.Fatalf("no authenticated session should not be rejected: %v", err)
    }
    if IsTrackedInbound(context.Background()) {
        t.Fatal("unattributed context was treated as a tracked session")
    }
    for _, raw := range []string{
        ",vless-share", "vless-share,", "vless-share,vless-share",
        "vless-share, vless-cdn", "vless-share,xmesh_stats_in!",
        "vless-share/other", "vless-share\n", "vless-share,vless-cdn,",
    } {
        if _, err := parseTrackedInboundTags(raw); err == nil {
            t.Fatalf("invalid coverage list accepted: %q", raw)
        }
    }
    fallback, err := parseTrackedInboundTags("")
    if err != nil || len(fallback) != 1 {
        t.Fatal("legacy isolated test inbound was not preserved")
    }
    if _, ok := fallback["vless-share"]; !ok {
        t.Fatal("default test inbound missing")
    }
}

func resetGate() {
    gate.Lock()
    defer gate.Unlock()
    gate.users = map[string]*userState{}
    gate.sessions = map[uint64]*trackedSession{}
}

func ctx(id uint32) context.Context {
    return c.ContextWithID(context.Background(),c.ID(id))
}

func TestSignedUserIdentityAcrossMainAndCDNEmailsCannotBypassFence(t *testing.T) {
    resetGate()
    mainEmail := "xmesh-u101--vless-main"
    cdnEmail := "xmesh-u101--vless-cdn"
    if canonicalMeterUser(mainEmail) != canonicalMeterUser(cdnEmail) ||
        canonicalMeterUser(mainEmail) != "xmesh-u101" {
        t.Fatal("signed same-user accounting identities not combined")
    }
    for _, tag := range []string{"xmesh-u101--PL%2FCDN", "xmesh-u101--🇵🇱CDN"} {
        if canonicalMeterUser(tag) != "xmesh-u101" {
            t.Fatalf("signed encoded transport tag escaped user fence: %s",tag)
        }
    }
    for _, malformed := range []string{"xmesh-u01--vless-main", "xmesh-u0--cdn",
        "xmesh-u101--", "xmesh-u101--invalid>tag", "xmesh-u102--cdn"} {
        if canonicalMeterUser(malformed) == canonicalMeterUser(mainEmail) {
            t.Fatalf("an unrelated or invalid identity was merged: %s",malformed)
        }
    }
    primary, secondary := ctx(2801),ctx(2802)
    stopped := make(chan string, 2)
    if err := Register(primary,mainEmail,func(){
        Finish(primary)
        RouteFinish(primary)
        stopped <- "main"
    }); err != nil { t.Fatal(err) }
    if err := Register(secondary,cdnEmail,func(){
        Finish(secondary)
        RouteFinish(secondary)
        stopped <- "cdn"
    }); err != nil { t.Fatal(err) }
    if err := BeginMeterWrite(cdnEmail); err != nil { t.Fatal(err) }
    outcome := make(chan result,1)
    failure := make(chan error,1)
    go func(){
        deadline,cancel := context.WithTimeout(context.Background(),2*time.Second)
        defer cancel()
        value,err := fence(deadline,mainEmail)
        if err != nil { failure <- err; return }
        outcome <- value
    }()
    <-stopped
    <-stopped
    for _, email := range []string{mainEmail,cdnEmail} {
        if err := Register(ctx(2803),email,func(){}); !errors.Is(err,errFenced) {
            t.Fatalf("revoked tag %s admitted another stream: %v",email,err)
        }
        if err := BeginMeterWrite(email); !errors.Is(err,errFenced) {
            t.Fatalf("revoked tag %s admitted another billable write: %v",email,err)
        }
        if BeginMeterRead(email) {
            t.Fatalf("revoked tag %s admitted another async meter read",email)
        }
    }
    select {
    case <-outcome: t.Fatal("fence completed before CDN writer finished")
    case err := <-failure: t.Fatalf("premature error: %v",err)
    case <-time.After(35*time.Millisecond):
    }
    EndMeterWrite(cdnEmail)
    select {
    case value := <-outcome:
        if !value.Fenced || !value.Drained || value.Active!=0 ||
           value.FinalMeterVerified || value.Refundable {
            t.Fatalf("incorrect canonical-user fence: %+v",value)
        }
    case err := <-failure:
        t.Fatal(err)
    case <-time.After(time.Second):
        t.Fatal("canonical-user fence did not finish")
    }
}

func TestSelectiveFenceAndRepeatNeverRefund(t *testing.T) {
    resetGate()
    first, second := ctx(11), ctx(12)
    stopped := make(chan struct{})
    if err := Register(first, "user-a", func() {
        Finish(first)
        RouteFinish(first)
        close(stopped)
    }); err != nil { t.Fatal(err) }
    if err := Register(second, "user-b", func() {}); err != nil { t.Fatal(err) }
    deadline,cancel := context.WithTimeout(context.Background(),2*time.Second)
    defer cancel()
    info,err := fence(deadline,"user-a")
    if err != nil { t.Fatal(err) }
    <-stopped
    if !info.Fenced || !info.Drained || info.Active != 0 ||
        info.FinalMeterVerified || info.Refundable {
        t.Fatalf("fake final meter: %+v",info)
    }
    if err := Register(ctx(13),"user-a",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("reopened user after fence: %v",err)
    }
    if err := Register(ctx(14),"user-b",func(){}); err != nil {
        t.Fatalf("unrelated user was blocked: %v",err)
    }
    again,err := fence(deadline,"user-a")
    if err != nil || again != info { t.Fatalf("non-idempotent: %+v %v",again,err) }
    Finish(second)
    RouteFinish(second)
    Finish(ctx(14))
    RouteFinish(ctx(14))
}

func TestFenceTimeoutKeepsAdmissionClosed(t *testing.T) {
    resetGate()
    s := ctx(15)
    if err:=Register(s,"stuck",func(){}); err!=nil { t.Fatal(err) }
    deadline,cancel:=context.WithTimeout(context.Background(),20*time.Millisecond)
    defer cancel()
    info,err:=fence(deadline,"stuck")
    if !errors.Is(err,context.DeadlineExceeded) || info.Refundable {
        t.Fatalf("timeout did not fail closed: %+v %v",info,err)
    }
    if err:=Register(ctx(16),"stuck",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("timeout resurrected admission: %v",err)
    }
    Finish(s)
    RouteFinish(s)
    recovered,err:=fence(context.Background(),"stuck")
    if err!=nil || recovered.FinalMeterVerified || recovered.Refundable ||
        !recovered.Drained {
        t.Fatalf("failed late completion: %+v %v",recovered,err)
    }
}

func TestManyConcurrentRegistrationsCannotCrossFence(t *testing.T) {
    resetGate()
    const count=180
    var wg sync.WaitGroup
    var mu sync.Mutex
    ids:=make([]uint32,0,count)
    for i:=0;i<count;i++ {
        wg.Add(1)
        go func(i int) {
            defer wg.Done()
            id:=uint32(i+100)
            c:=ctx(id)
            if err:=Register(c,"crowd",func(){Finish(c);RouteFinish(c)}); err==nil {
                mu.Lock()
                ids=append(ids,id)
                mu.Unlock()
            } else if !errors.Is(err,errFenced) {
                t.Errorf("invalid admission failure: %v",err)
            }
        }(i)
    }
    stopper := make(chan struct{})
    go func() {
        deadline,cancel:=context.WithTimeout(context.Background(),3*time.Second)
        defer cancel()
        info,err:=fence(deadline,"crowd")
        if err!=nil || !info.Fenced || !info.Drained || info.Active!=0 ||
            info.Refundable || info.FinalMeterVerified {
            t.Errorf("incorrect concurrent fence result: %+v %v",info,err)
        }
        close(stopper)
    }()
    wg.Wait()
    <-stopper
    mu.Lock()
    admitted:=len(ids)
    mu.Unlock()
    if admitted>count { t.Fatalf("admission count exceeded: %d",admitted) }
    if err:=Register(ctx(300),"crowd",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("concurrent re-entry: %v",err)
    }
}

func TestDuplicateLiveSessionIDCannotFakeDrain(t *testing.T) {
    resetGate()
    s := ctx(19)
    if err := Register(s,"owner",func(){}); err != nil { t.Fatal(err) }
    if err := Register(s,"owner",func(){Finish(s)}); !errors.Is(err,errDuplicateSession) {
        t.Fatalf("second live stream silently shared completion token: %v",err)
    }
    short,cancel := context.WithTimeout(context.Background(),20*time.Millisecond)
    defer cancel()
    snapshot,err := fence(short,"owner")
    if !errors.Is(err,context.DeadlineExceeded) || snapshot.Refundable {
        t.Fatalf("duplicate produced false terminal proof: %+v %v",snapshot,err)
    }
    Finish(s)
    RouteFinish(s)
    recovered,err := fence(context.Background(),"owner")
    if err!=nil || !recovered.Drained || recovered.FinalMeterVerified ||
        recovered.Refundable { t.Fatalf("invalid finality: %+v %v",recovered,err) }
    if err := Register(s,"owner",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("fence allowed reentry: %v",err)
    }
}

func TestWorkerExitCannotHideRunningRoute(t *testing.T) {
    resetGate()
    s := ctx(20)
    if err := Register(s,"route-live",func(){Finish(s)}); err != nil {
        t.Fatal(err)
    }
    // A returned inbound worker does not imply routedDispatch finished.
    Finish(s)
    short,cancel := context.WithTimeout(context.Background(),20*time.Millisecond)
    defer cancel()
    snapshot,err := fence(short,"route-live")
    if !errors.Is(err,context.DeadlineExceeded) || snapshot.Drained || snapshot.Refundable {
        t.Fatalf("premature drain before route completion: %+v %v",snapshot,err)
    }
    if err:=Register(ctx(21),"route-live",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("waiting for route reopened admission: %v",err)
    }
    RouteFinish(s)
    recovered,err := fence(context.Background(),"route-live")
    if err != nil || !recovered.Drained || recovered.FinalMeterVerified ||
        recovered.Refundable { t.Fatalf("false finality: %+v %v",recovered,err) }
}

func TestMeterWriteBarrierPreventsLateWrites(t *testing.T) {
    resetGate()
    s:=ctx(201)
    stopped:=make(chan struct{})
    if err:=Register(s,"metered",func(){close(stopped)});err!=nil{t.Fatal(err)}
    if err:=BeginMeterWrite("metered");err!=nil{t.Fatal(err)}
    outcome:=make(chan struct{
        info result
        err error
    },1)
    go func(){
        ctx,cancel:=context.WithTimeout(context.Background(),time.Second)
        defer cancel()
        info,err:=fence(ctx,"metered")
        outcome<-struct{info result;err error}{info,err}
    }()
    <-stopped
    Finish(s)
    RouteFinish(s)
    if err:=BeginMeterWrite("metered");!errors.Is(err,errFenced) {
        t.Fatalf("fence allowed new meter operation: %v",err)
    }
    select {
    case got:=<-outcome:
        t.Fatalf("drained with outstanding meter write: %+v",got)
    case <-time.After(40*time.Millisecond):
    }
    EndMeterWrite("metered")
    select {
    case got:=<-outcome:
        if got.err!=nil||!got.info.Drained||got.info.FinalMeterVerified||got.info.Refundable {
            t.Fatalf("invalid complete gate: %+v",got)
        }
    case <-time.After(2*time.Second):
        t.Fatal("fence did not resolve after write completed")
    }
    if err:=BeginMeterWrite("metered");!errors.Is(err,errFenced) {
        t.Fatalf("new metered write accepted after completion: %v",err)
    }
}

func TestMeterReadBarrierIncludesTimedOutAsyncReader(t *testing.T){
    resetGate()
    stream:=ctx(302)
    stopped:=make(chan struct{})
    if err:=Register(stream,"pending-read",func(){close(stopped)});err!=nil{
        t.Fatal(err)
    }
    if !BeginMeterRead("pending-read"){t.Fatal("failed to start test read")}
    outcome:=make(chan struct{info result;err error},1)
    go func(){
        bound,cancel:=context.WithTimeout(context.Background(),time.Second)
        defer cancel()
        info,err:=fence(bound,"pending-read")
        outcome<-struct{info result;err error}{info,err}
    }()
    <-stopped
    Finish(stream)
    RouteFinish(stream)
    if BeginMeterRead("pending-read"){
        t.Fatal("revocation allowed a new metered read")
    }
    select {
    case got:=<-outcome:
        t.Fatalf("premature drain while timed-out read pending: %+v",got)
    case <-time.After(50*time.Millisecond):
    }
    EndMeterRead("pending-read")
    select {
    case got:=<-outcome:
        if got.err!=nil||!got.info.Drained||got.info.FinalMeterVerified||
            got.info.Refundable {
            t.Fatalf("completion falsely authorizes refundable bytes: %+v",got)
        }
    case <-time.After(2*time.Second):
        t.Fatal("drain never completed after pending read was accounted")
    }
    if BeginMeterRead("pending-read"){
        t.Fatal("read admission reopened after drain")
    }
}
