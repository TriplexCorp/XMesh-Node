func TestMain(m *testing.M) {
    // Unit tests exercise only the in-memory gate. Never start the privileged
    // Unix control endpoint from these tests.
    enabled = true
    os.Exit(m.Run())
}

func resetGate() {
    gate.Lock()
    defer gate.Unlock()
    gate.users = map[string]*userState{}
    gate.sessions = map[uint64]*trackedSession{}
}

func ctx(id uint32) context.Context {
    return c.ContextWithID(context.Background(),c.ID(id))
}

func TestSelectiveFenceAndRepeatNeverRefund(t *testing.T) {
    resetGate()
    first, second := ctx(11), ctx(12)
    stopped := make(chan struct{})
    if err := Register(first, "user-a", func() {
        Finish(first)
        RouteFinish(first)
        close(stopped)
    }); err != nil { t.Fatal(err) }
    if err := Register(second, "user-b", func() {}); err != nil { t.Fatal(err) }
    deadline,cancel := context.WithTimeout(context.Background(),2*time.Second)
    defer cancel()
    info,err := fence(deadline,"user-a")
    if err != nil { t.Fatal(err) }
    <-stopped
    if !info.Fenced || !info.Drained || info.Active != 0 ||
        info.FinalMeterVerified || info.Refundable {
        t.Fatalf("fake final meter: %+v",info)
    }
    if err := Register(ctx(13),"user-a",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("reopened user after fence: %v",err)
    }
    if err := Register(ctx(14),"user-b",func(){}); err != nil {
        t.Fatalf("unrelated user was blocked: %v",err)
    }
    again,err := fence(deadline,"user-a")
    if err != nil || again != info { t.Fatalf("non-idempotent: %+v %v",again,err) }
    Finish(second)
    RouteFinish(second)
    Finish(ctx(14))
    RouteFinish(ctx(14))
}

func TestFenceTimeoutKeepsAdmissionClosed(t *testing.T) {
    resetGate()
    s := ctx(15)
    if err:=Register(s,"stuck",func(){}); err!=nil { t.Fatal(err) }
    deadline,cancel:=context.WithTimeout(context.Background(),20*time.Millisecond)
    defer cancel()
    info,err:=fence(deadline,"stuck")
    if !errors.Is(err,context.DeadlineExceeded) || info.Refundable {
        t.Fatalf("timeout did not fail closed: %+v %v",info,err)
    }
    if err:=Register(ctx(16),"stuck",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("timeout resurrected admission: %v",err)
    }
    Finish(s)
    RouteFinish(s)
    recovered,err:=fence(context.Background(),"stuck")
    if err!=nil || recovered.FinalMeterVerified || recovered.Refundable ||
        !recovered.Drained {
        t.Fatalf("failed late completion: %+v %v",recovered,err)
    }
}

func TestManyConcurrentRegistrationsCannotCrossFence(t *testing.T) {
    resetGate()
    const count=180
    var wg sync.WaitGroup
    var mu sync.Mutex
    ids:=make([]uint32,0,count)
    for i:=0;i<count;i++ {
        wg.Add(1)
        go func(i int) {
            defer wg.Done()
            id:=uint32(i+100)
            c:=ctx(id)
            if err:=Register(c,"crowd",func(){Finish(c);RouteFinish(c)}); err==nil {
                mu.Lock()
                ids=append(ids,id)
                mu.Unlock()
            } else if !errors.Is(err,errFenced) {
                t.Errorf("invalid admission failure: %v",err)
            }
        }(i)
    }
    stopper := make(chan struct{})
    go func() {
        deadline,cancel:=context.WithTimeout(context.Background(),3*time.Second)
        defer cancel()
        info,err:=fence(deadline,"crowd")
        if err!=nil || !info.Fenced || !info.Drained || info.Active!=0 ||
            info.Refundable || info.FinalMeterVerified {
            t.Errorf("incorrect concurrent fence result: %+v %v",info,err)
        }
        close(stopper)
    }()
    wg.Wait()
    <-stopper
    mu.Lock()
    admitted:=len(ids)
    mu.Unlock()
    if admitted>count { t.Fatalf("admission count exceeded: %d",admitted) }
    if err:=Register(ctx(300),"crowd",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("concurrent re-entry: %v",err)
    }
}

func TestDuplicateLiveSessionIDCannotFakeDrain(t *testing.T) {
    resetGate()
    s := ctx(19)
    if err := Register(s,"owner",func(){}); err != nil { t.Fatal(err) }
    if err := Register(s,"owner",func(){Finish(s)}); !errors.Is(err,errDuplicateSession) {
        t.Fatalf("second live stream silently shared completion token: %v",err)
    }
    short,cancel := context.WithTimeout(context.Background(),20*time.Millisecond)
    defer cancel()
    snapshot,err := fence(short,"owner")
    if !errors.Is(err,context.DeadlineExceeded) || snapshot.Refundable {
        t.Fatalf("duplicate produced false terminal proof: %+v %v",snapshot,err)
    }
    Finish(s)
    RouteFinish(s)
    recovered,err := fence(context.Background(),"owner")
    if err!=nil || !recovered.Drained || recovered.FinalMeterVerified ||
        recovered.Refundable { t.Fatalf("invalid finality: %+v %v",recovered,err) }
    if err := Register(s,"owner",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("fence allowed reentry: %v",err)
    }
}

func TestWorkerExitCannotHideRunningRoute(t *testing.T) {
    resetGate()
    s := ctx(20)
    if err := Register(s,"route-live",func(){Finish(s)}); err != nil {
        t.Fatal(err)
    }
    // A returned inbound worker does not imply routedDispatch finished.
    Finish(s)
    short,cancel := context.WithTimeout(context.Background(),20*time.Millisecond)
    defer cancel()
    snapshot,err := fence(short,"route-live")
    if !errors.Is(err,context.DeadlineExceeded) || snapshot.Drained || snapshot.Refundable {
        t.Fatalf("premature drain before route completion: %+v %v",snapshot,err)
    }
    if err:=Register(ctx(21),"route-live",func(){}); !errors.Is(err,errFenced) {
        t.Fatalf("waiting for route reopened admission: %v",err)
    }
    RouteFinish(s)
    recovered,err := fence(context.Background(),"route-live")
    if err != nil || !recovered.Drained || recovered.FinalMeterVerified ||
        recovered.Refundable { t.Fatalf("false finality: %+v %v",recovered,err) }
}

func TestMeterWriteBarrierPreventsLateWrites(t *testing.T) {
    resetGate()
    s:=ctx(201)
    stopped:=make(chan struct{})
    if err:=Register(s,"metered",func(){close(stopped)});err!=nil{t.Fatal(err)}
    if err:=BeginMeterWrite("metered");err!=nil{t.Fatal(err)}
    outcome:=make(chan struct{
        info result
        err error
    },1)
    go func(){
        ctx,cancel:=context.WithTimeout(context.Background(),time.Second)
        defer cancel()
        info,err:=fence(ctx,"metered")
        outcome<-struct{info result;err error}{info,err}
    }()
    <-stopped
    Finish(s)
    RouteFinish(s)
    if err:=BeginMeterWrite("metered");!errors.Is(err,errFenced) {
        t.Fatalf("fence allowed new meter operation: %v",err)
    }
    select {
    case got:=<-outcome:
        t.Fatalf("drained with outstanding meter write: %+v",got)
    case <-time.After(40*time.Millisecond):
    }
    EndMeterWrite("metered")
    select {
    case got:=<-outcome:
        if got.err!=nil||!got.info.Drained||got.info.FinalMeterVerified||got.info.Refundable {
            t.Fatalf("invalid complete gate: %+v",got)
        }
    case <-time.After(2*time.Second):
        t.Fatal("fence did not resolve after write completed")
    }
    if err:=BeginMeterWrite("metered");!errors.Is(err,errFenced) {
        t.Fatalf("new metered write accepted after completion: %v",err)
    }
}

func TestMeterReadBarrierIncludesTimedOutAsyncReader(t *testing.T){
    resetGate()
    stream:=ctx(302)
    stopped:=make(chan struct{})
    if err:=Register(stream,"pending-read",func(){close(stopped)});err!=nil{
        t.Fatal(err)
    }
    if !BeginMeterRead("pending-read"){t.Fatal("failed to start test read")}
    outcome:=make(chan struct{info result;err error},1)
    go func(){
        bound,cancel:=context.WithTimeout(context.Background(),time.Second)
        defer cancel()
        info,err:=fence(bound,"pending-read")
        outcome<-struct{info result;err error}{info,err}
    }()
    <-stopped
    Finish(stream)
    RouteFinish(stream)
    if BeginMeterRead("pending-read"){
        t.Fatal("revocation allowed a new metered read")
    }
    select {
    case got:=<-outcome:
        t.Fatalf("premature drain while timed-out read pending: %+v",got)
    case <-time.After(50*time.Millisecond):
    }
    EndMeterRead("pending-read")
    select {
    case got:=<-outcome:
        if got.err!=nil||!got.info.Drained||got.info.FinalMeterVerified||
            got.info.Refundable {
            t.Fatalf("completion falsely authorizes refundable bytes: %+v",got)
        }
    case <-time.After(2*time.Second):
        t.Fatal("drain never completed after pending read was accounted")
    }
    if BeginMeterRead("pending-read"){
        t.Fatal("read admission reopened after drain")
    }
}
